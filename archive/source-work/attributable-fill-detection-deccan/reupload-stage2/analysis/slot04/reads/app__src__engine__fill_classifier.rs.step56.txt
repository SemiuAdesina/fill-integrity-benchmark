241	    /// Divide one block match across the orders it covers.
242	    ///
243	    /// The venue reports only a total and names the orders it was matched
244	    /// against; the division rule is in `/app/spec/venue-semantics.md`. Every step
245	    /// is integer arithmetic, and any part of the block that no tracked order can
246	    /// account for is dropped rather than assigned.
247	    fn absorb_block(&mut self, block: BlockFill) {
248	        // The same match, redelivered, is the same match. `block_ref` is the
249	        // idempotency key, so this is what stops a block being counted twice.
250	        if !self.absorbed_blocks.insert(block.block_ref.clone()) {
251	            return;
252	        }
253	
254	        // Every tracked order the block names, resolved through the match time:
255	        // the venue recycles identifiers, so the window is what decides which of
256	        // this engine's orders an id is naming here.
257	        let mut covered: Vec<usize> = Vec::new();
258	        for order_id in &block.covers {
259	            let Some(index) = self.index_of_order_at(order_id, block.at) else {
260	                continue;
261	            };
262	            if !covered.contains(&index) {
263	                covered.push(index);
264	            }
265	        }
266	
267	        // Eligible set: window containing the block's match time, and still
268	        // short. An order the venue has already closed has nothing left
269	        // outstanding, so it contributes no basis and receives no share.
270	        let mut eligible: Vec<usize> = Vec::new();
271	        for index in &covered {
272	            let record = &self.submissions[*index];
273	            if record.finalized {
274	                continue;
275	            }
276	            if !Self::in_window(record, block.at) {
277	                continue;
278	            }
279	            if record.outstanding() <= 0 {
280	                continue;
281	            }
282	            eligible.push(*index);
283	        }
284	
285	        // `covers` arrives in whatever order the venue happened to build it and
286	        // the outcome must not depend on that, so the eligible set is put in the
287	        // order the specification states before anything is divided: window
288	        // start, then venue sequence, then order id.
289	        eligible.sort_by(|left, right| {
290	            Self::division_order(&self.submissions[*left])
291	                .cmp(&Self::division_order(&self.submissions[*right]))
292	        });
293	
294	        let outstanding: Vec<Qty> = eligible
295	            .iter()
296	            .map(|index| self.submissions[*index].outstanding())
297	            .collect();
298	
299	        // Quantity above the basis belongs to orders this engine does not track.
300	        let basis: Qty = outstanding.iter().sum();
301	        if basis <= 0 {
302	            self.finalize(&covered);
303	            return;
304	        }
305	        let divisible = if block.block_qty < basis {
306	            block.block_qty
307	        } else {
308	            basis
309	        };
310	
311	        let mut shares: Vec<Qty> = outstanding
312	            .iter()
313	            .map(|outstanding| proportional_share(divisible, *outstanding, basis))
314	            .collect();
315	
316	        // Truncation leaves a few base units unassigned. The first `r` orders in
317	        // the ordering above take one each — not the largest order, and not the
318	        // order with the largest truncated fraction.
319	        let assigned: Qty = shares.iter().sum();
320	        let mut remainder = divisible - assigned;
321	        for share in shares.iter_mut() {
322	            if remainder <= 0 {
323	                break;
324	            }
325	            *share += 1;
326	            remainder -= 1;
327	        }
328	
329	        for (index, share) in eligible.iter().zip(shares) {
330	            self.submissions[*index].block_filled += share;
331	        }
332	
333	        self.finalize(&covered);
334	    }
