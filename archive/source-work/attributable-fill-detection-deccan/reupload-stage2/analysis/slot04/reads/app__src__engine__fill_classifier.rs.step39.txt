189	    pub fn absorb_batch(&mut self, batch: FillBatch) -> usize {
190	        let count = batch.events.len();
191	        for event in batch.events {
192	            self.absorb_event(event);
193	        }
194	        for block in batch.blocks {

196	        }

201	    fn absorb_event(&mut self, event: FillEvent) {
202	        let Some(index) = self.locate(&event.order_id, &event.client_ref) else {
203	            return;
204	        };
205	        // The window is tested against the venue's own match timestamp, not the
206	        // cycle that happened to read the event: the feed lags and replays.
207	        if !Self::in_window(&self.submissions[index], event.at) {
208	            return;
209	        }
210	        let record = &mut self.submissions[index];
211	        if record.order_id.is_none() {
212	            record.order_id = Some(event.order_id.clone());
213	        }
214	        // `cumulative_filled_qty` is the order's running total, not the size of
215	        // this match, so it raises a high-water mark rather than adding to one.
216	        // The feed delivers at least once, and both a replay and an out-of-order
217	        // arrival therefore have to leave the figure where it is.
218	        if event.cumulative_filled_qty > record.direct_filled {
219	            record.direct_filled = event.cumulative_filled_qty;
220	        }
221	    }

224	    ///
225	    /// The venue reports only a total and names the orders it was matched
226	    /// against; the division rule is in `/app/spec/venue-semantics.md`. Every step
227	    /// is integer arithmetic, and any part of the block that no tracked order can
228	    /// account for is dropped rather than assigned.
229	    fn absorb_block(&mut self, block: BlockFill) {
230	        // The same match, redelivered, is the same match. `block_ref` is the
231	        // idempotency key, so this is what stops a block being counted twice.
232	        if !self.absorbed_blocks.insert(block.block_ref.clone()) {
233	            return;
234	        }

237	        // still short. An order the venue has already closed has nothing left
238	        // outstanding, so it contributes no basis and receives no share.
239	        let mut eligible: Vec<usize> = Vec::new();
240	        for order_id in &block.covers {
241	            let Some(index) = self.index_of_order(order_id) else {
242	                continue;
243	            };
244	            let record = &self.submissions[index];
245	            if record.finalized {
246	                continue;
247	            }
248	            if !Self::in_window(record, block.at) {
249	                continue;
250	            }
251	            if record.outstanding() <= 0 {
252	                continue;

254	            eligible.push(index);
255	        }

264	        });
