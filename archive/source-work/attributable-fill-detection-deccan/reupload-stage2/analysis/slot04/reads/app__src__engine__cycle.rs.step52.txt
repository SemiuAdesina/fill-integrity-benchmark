332	    /// committed to for this leg.
333	    ///
334	    /// This is the term that stops consecutive cycles from sending the same
335	    /// exposure twice: a commitment counts as working until it resolves.
336	    fn working_for(&self, leg: &PendingOrder) -> Qty {
337	        let mut total = 0;
338	        for record in self.records_for(leg) {
339	            total += match record.order_id {
340	                // The venue has named it, so it contributes whatever it has
341	                // still to acquire. One the venue has since closed contributes
342	                // nothing, whatever it managed to fill: the venue cancelled the
343	                // rest and the gap it leaves behind is real.
344	                Some(_) if self.ledger.has(&record.client_ref) => {
345	                    self.ledger.outstanding_for(&record.client_ref)
346	                }
347	                Some(_) => {
348	                    if record.size > 0 {
349	                        record.size
350	                    } else {
351	                        0
352	                    }
353	                }
354	                // The acknowledgement was lost. The venue may have created the
355	                // order and the engine cannot see it, so its full size counts
356	                // as working rather than nothing at all.
357	                None => {
358	                    if record.size > 0 {
359	                        record.size
360	                    } else {
361	                        0
362	                    }
363	                }
364	            };
365	        }
366	        total
367	    }
