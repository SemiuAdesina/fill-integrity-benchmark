175	    /// Match up corrective orders whose fate was unknown when they were sent.
176	    ///
177	    /// A record without an `order_id` is one the venue never acknowledged. The
178	    /// client reference is the venue's idempotency key, so resubmitting under it
179	    /// returns the original order rather than creating a second one.
180	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
181	        // Identity is recovered when a later observation carries that
182	        // `client_ref`: the ledger learns the venue id from the open-order list
183	        // or from a fill event, and the record catches up with it here.
184	        let recovered: Vec<(ClientRef, OrderId)> = self
185	            .neutralizations
186	            .iter()
187	            .filter(|record| record.order_id.is_none())
188	            .filter_map(|record| {
189	                self.ledger
190	                    .order_for(&record.client_ref)
191	                    .map(|order_id| (record.client_ref.clone(), order_id))
192	            })
193	            .collect();
194	        for (client_ref, order_id) in recovered {
195	            if let Some(slot) = self
196	                .neutralizations
197	                .iter_mut()
198	                .find(|slot| slot.client_ref == client_ref && slot.order_id.is_none())
199	            {
200	                slot.order_id = Some(order_id);
201	            }
202	        }
203	
204	        let unconfirmed: Vec<NeutralizationRecord> = self
205	            .neutralizations
206	            .iter()
207	            .filter(|record| record.order_id.is_none())
208	            .cloned()
209	            .collect();
210	        for record in unconfirmed {
211	            if !self
212	                .attempted_this_cycle
213	                .insert((record.strategy_id.clone(), record.venue))
214	            {
215	                continue;
216	            }
217	            let request = Self::neutralization_request(&record);
218	            let size = record.size;
219	            let client_ref = record.client_ref.clone();
220	            match self.gateway.submit(&request) {
221	                // The venue named the order, whether it is the one created by the
222	                // first attempt or one this retry created. Either way there is
223	                // exactly one economic order behind this client reference.
224	                Ok(accepted) => {
225	                    self.ledger.track(
226	                        client_ref.clone(),
227	                        Some(accepted.order_id.clone()),
228	                        Some(accepted.venue_seq),
229	                        size,
230	                        now,
231	                    );
232	                    if let Some(slot) = self
233	                        .neutralizations
234	                        .iter_mut()
235	                        .find(|slot| slot.client_ref == client_ref)
236	                    {
237	                        slot.order_id = Some(accepted.order_id);
238	                    }
239	                }
240	                // Still unresolved. Keep the record and its client reference so a
241	                // later cycle retries under the same one.
242	                Err(SubmitError::AcknowledgementLost) => {}
243	                // Nothing was created by that attempt, so there is nothing to
244	                // wait for. Drop the record and leave the exposure open for a
245	                // later cycle to recompute and try again.
246	                Err(SubmitError::Rejected(_)) | Err(SubmitError::Transport(_)) => {
247	                    self.neutralizations.retain(|slot| slot.client_ref != client_ref);
248	                }
249	            }
250	        }
251	    }
