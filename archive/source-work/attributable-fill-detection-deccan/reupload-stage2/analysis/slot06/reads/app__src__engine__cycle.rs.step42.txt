427	    /// the next one takes it out of the book before covering what is still

431	    ///
432	    /// A leg whose state is momentarily unknown is never neutralized against:
433	    /// covering quantity the desk may already own doubles the very exposure the
434	    /// correction was meant to close.
435	    fn close_gap(&mut self, leg: &PendingOrder, now: Timestamp) -> Result<(), PortError> {
436	        match self.visibility(leg) {
437	            LegVisibility::Resting => {
438	                // Mark it now; cancel and cover from the next cycle onwards.
439	                if self.closing.insert(leg.order_id.clone()) {
440	                    return Ok(());
441	                }
442	                if self.gateway.cancel(&leg.order_id).is_err() {
443	                    return Ok(());
444	                }
445	                self.resting_at_venue.remove(&leg.order_id);
446	                self.replaced.insert(leg.order_id.clone());
447	            }
448	            // Not resting at the venue, and its state is known, so there is
449	            // nothing to cancel first.
450	            LegVisibility::Withdrawn | LegVisibility::Closed => {}
451	            // Unknown: it may still be resting and may still fill.
452	            LegVisibility::Unconfirmed => return Ok(()),
453	        }
454	
455	        // Only the residual is covered, net of everything already attributed and
456	        // everything already working, so consecutive cycles cannot send the same
457	        // exposure twice.
458	        let residual = self.residual_for(leg);
459	        if residual <= 0 {
460	            return Ok(());
461	        }
462	        let plan = match plan_neutralization(
463	            self.gateway,
464	            leg,
465	            residual,
466	            self.config.neutralization_slippage,
467	        )? {
468	            Some(plan) => plan,
469	            None => return Ok(()),
470	        };

473	        // outcome is unknown is still something this engine knows it committed to.
474	        let client_ref = self.next_client_ref();

478	            venue: plan.venue,
479	            instrument_id: plan.instrument_id,

481	            order_id: None,
482	            limit_price: plan.limit_price,
483	            size,
484	            expiry_time: leg.expiry_time,
485	        };
486	        self.ledger
487	            .track(client_ref.clone(), None, None, size, now);
488	        self.neutralizations.push(record.clone());

494	                    client_ref.clone(),
495	                    Some(accepted.order_id.clone()),
496	                    Some(accepted.venue_seq),
497	                    size,
498	                    now,
499	                );
