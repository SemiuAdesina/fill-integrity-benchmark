166	    /// Match up corrective orders whose fate was unknown when they were sent.
167	    ///
168	    /// A record without an `order_id` is one the venue never acknowledged. The
169	    /// client reference is the venue's idempotency key, so resubmitting under it
170	    /// returns the original order rather than creating a second one, and a retry
171	    /// under a freshly minted reference would ask for the exposure twice.
172	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
173	        // Identity is recovered when a later observation carries the reference.
174	        let mut recovered: Vec<(ClientRef, OrderId)> = Vec::new();
175	        for record in self.neutralizations.iter() {
176	            if record.order_id.is_some() {
177	                continue;
178	            }
179	            if let Some(order_id) = self.ledger.order_for(&record.client_ref) {
180	                recovered.push((record.client_ref.clone(), order_id));
181	            }
182	        }
183	        for (client_ref, order_id) in recovered {
184	            if let Some(slot) = self
185	                .neutralizations
186	                .iter_mut()
187	                .find(|slot| slot.client_ref == client_ref)
188	            {
189	                slot.order_id = Some(order_id);
190	            }
191	        }
192	
193	        let unconfirmed: Vec<NeutralizationRecord> = self
194	            .neutralizations
195	            .iter()
196	            .filter(|record| record.order_id.is_none())
197	            .cloned()
198	            .collect();
199	        for record in unconfirmed {
200	            let request = Self::neutralization_request(&record);
201	            match self.gateway.submit(&request) {
202	                Ok(accepted) => {
203	                    self.ledger.track(
204	                        record.client_ref.clone(),
205	                        Some(accepted.order_id.clone()),
206	                        Some(accepted.venue_seq),
207	                        record.size,
208	                        now,
209	                    );
210	                    if let Some(slot) = self
211	                        .neutralizations
212	                        .iter_mut()
213	                        .find(|slot| slot.client_ref == record.client_ref)
214	                    {
215	                        slot.order_id = Some(accepted.order_id);
216	                    }
217	                }
218	                // The venue may have created it. Keep the record and the
219	                // reference and try again on a later cycle.
220	                Err(SubmitError::AcknowledgementLost) => {}
221	                // Nothing was created by this attempt, so there is nothing to
222	                // keep: drop the commitment and let a later cycle recompute the
223	                // residual and try again.
224	                Err(_) => {
225	                    self.neutralizations
226	                        .retain(|slot| slot.client_ref != record.client_ref);
227	                    self.ledger.discard(&record.client_ref);
228	                }
229	            }
230	        }
231	    }

246	    fn enter_new_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
247	        let available_cash = self.inventory.settlement_balance()?;
248	        let candidates = self.strategies.candidates()?;
249	        for candidate in candidates {
250	            if self.positions.has(&candidate.strategy_id) {
251	                continue;
252	            }
253	            if self.attempted.contains(&candidate.strategy_id) {
254	                continue;
255	            }
256	            if candidate.pair_cost() > self.config.max_pair_cost {
257	                continue;
258	            }
259	            if now >= candidate.expiry_time {
260	                continue;
261	            }
262	            let north_size = leg_size(candidate.north_depth, self.config.max_leg_size);
263	            let south_size = leg_size(candidate.south_depth, self.config.max_leg_size);
264	            if north_size < self.config.min_leg_size || south_size < self.config.min_leg_size {
265	                continue;
266	            }
267	            let notional = (candidate.north_price * north_size as f64)
268	                + (candidate.south_price * south_size as f64);
269	            if notional > available_cash {
270	                continue;
271	            }
272	            self.attempted.insert(candidate.strategy_id.clone());
273	            self.place_leg(&candidate, Venue::North, north_size, now);
274	            self.place_leg(&candidate, Venue::South, south_size, now);
275	        }
276	        Ok(())
277	    }
