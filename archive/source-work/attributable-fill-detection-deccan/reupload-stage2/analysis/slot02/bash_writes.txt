pub struct FakeGateway {
    failures: RefCell<Vec<(usize, SubmitError)>>,
    submits: Cell<usize>,
    pub submitted: RefCell<Vec<OrderRequest>>,
    pub open: RefCell<Vec<OpenOrder>>,
    pub cancelled: RefCell<Vec<OrderId>>,
    pub best_offer: Cell<Price>,
    next_seq: Cell<u64>,
}
impl FakeGateway {
    pub fn new() -> Self {
        Self { failures: RefCell::new(Vec::new()), submits: Cell::new(0),
            submitted: RefCell::new(Vec::new()), open: RefCell::new(Vec::new()),
            cancelled: RefCell::new(Vec::new()), best_offer: Cell::new(0.5), next_seq: Cell::new(100) }
    }
    pub fn fail_on(&self, submission: usize, error: SubmitError) {
        self.failures.borrow_mut().push((submission, error));
    }
    pub fn set_open(&self, open: Vec<OpenOrder>) { *self.open.borrow_mut() = open; }
    pub fn corrective(&self) -> Vec<(String, Qty)> {
        self.submitted.borrow().iter().filter(|r| r.kind == OrderKind::ImmediateOrCancel)
            .map(|r| (r.client_ref.to_string(), r.size)).collect()
    }
    pub fn resting(&self) -> Vec<(String, Qty)> {
        self.submitted.borrow().iter().filter(|r| r.kind == OrderKind::RestingLimit)
            .map(|r| (r.client_ref.to_string(), r.size)).collect()
    }
    pub fn cancels(&self) -> Vec<String> {
        self.cancelled.borrow().iter().map(|o| o.to_string()).collect()
    }
}
impl OrderGateway for FakeGateway {
    fn submit(&self, request: &OrderRequest) -> Result<OrderAccepted, SubmitError> {
        self.submitted.borrow_mut().push(request.clone());
        let position = self.submits.get() + 1;
        self.submits.set(position);
        let failure = { let mut f = self.failures.borrow_mut();
            f.iter().position(|(a, _)| *a == position).map(|i| f.remove(i).1) };
        if let Some(error) = failure { return Err(error); }
        let seq = self.next_seq.get() + 1;
        self.next_seq.set(seq);
        Ok(OrderAccepted { order_id: OrderId::new(format!("V-{seq}")), client_ref: request.client_ref.clone(), venue_seq: seq })
    }
    fn cancel(&self, order_id: &OrderId) -> Result<(), PortError> {
        self.cancelled.borrow_mut().push(order_id.clone()); Ok(())
    }
    fn open_orders(&self) -> Result<Vec<OpenOrder>, PortError> { Ok(self.open.borrow().clone()) }
    fn best_offer(&self, _i: &InstrumentId) -> Result<Option<Price>, PortError> { Ok(Some(self.best_offer.get())) }
}
