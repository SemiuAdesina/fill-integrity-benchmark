2	//! One engine cycle: read the venue, decide what completed, act, persist.
3	//!
4	//! `run_cycle` is the only entry point the graded suite calls. The order of the
5	//! steps below is the order they run in, and everything else in the crate is a
6	//! type or a helper reached from here.
7

26	/// What the live order list currently shows about a leg the engine tracks.
27	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28	enum LegVisibility {
29	    /// Present in `open_orders`.
30	    Resting,
31	    /// Absent from `open_orders`, and this engine cancelled it.
32	    Withdrawn,
33	    /// Absent from `open_orders`, and this engine did not cancel it.
34	    Unconfirmed,
35	}
36	
37	pub struct Engine<'a> {
38	    config: EngineConfig,
39	    strategies: &'a dyn StrategySource,
40	    inventory: &'a dyn InventorySource,
41	    gateway: &'a dyn OrderGateway,
42	    fills: &'a dyn FillEventSource,
43	    store: &'a dyn StateStore,
44	    clock: &'a dyn Clock,
45	    pending: PendingOrderManager,
46	    positions: PositionManager,
47	    attempted: BTreeSet<StrategyId>,
48	    /// Per-order attribution state, and the feed cursor that produced it.
49	    ledger: AttributionLedger,
50	    /// Corrective orders this engine has committed to, resolved or not.
51	    neutralizations: Vec<NeutralizationRecord>,
52	    /// Legs marked for closure on an earlier cycle.
53	    closing: BTreeSet<OrderId>,
54	    /// Legs this engine has cancelled at the venue.
55	    replaced: BTreeSet<OrderId>,
56	    /// Order ids the last `open_orders` call reported as live. Not durable: it is
57	    /// re-read every cycle and says nothing after a restart.
58	    resting_at_venue: BTreeSet<OrderId>,
59	    loaded: bool,
60	    client_seq: u64,
61	    observed_fill_events: usize,
62	}
63	
64	impl<'a> Engine<'a> {
65	    #[allow(clippy::too_many_arguments)]
66	    pub fn new(
67	        config: EngineConfig,
68	        strategies: &'a dyn StrategySource,
69	        inventory: &'a dyn InventorySource,
70	        gateway: &'a dyn OrderGateway,
71	        fills: &'a dyn FillEventSource,
72	        store: &'a dyn StateStore,
73	        clock: &'a dyn Clock,
74	    ) -> Self {
75	        Self {
76	            config,
77	            strategies,
78	            inventory,
79	            gateway,
80	            fills,
81	            store,
82	            clock,
83	            pending: PendingOrderManager::new(),
84	            positions: PositionManager::new(),
85	            attempted: BTreeSet::new(),
86	            ledger: AttributionLedger::new(),
87	            neutralizations: Vec::new(),
88	            closing: BTreeSet::new(),
89	            replaced: BTreeSet::new(),
90	            resting_at_venue: BTreeSet::new(),
91	            loaded: false,
92	            client_seq: 0,
93	            observed_fill_events: 0,
94	        }
95	    }
96	
97	    pub fn positions(&self) -> Vec<Position> {
98	        self.positions.all()
99	    }
100	
101	    pub fn pending_orders(&self) -> Vec<PendingOrder> {
102	        self.pending.all()
103	    }
104	
105	    pub fn observed_fill_events(&self) -> usize {
106	        self.observed_fill_events
107	    }
108	
109	    pub fn run_cycle(&mut self) -> Result<(), PortError> {
110	        self.ensure_loaded()?;
111	        let now = self.clock.now();
112	        self.sync_observations();
113	        self.reconcile_neutralizations(now);
114	        self.enter_new_strategies(now)?;
115	        self.reconcile_open_strategies(now)?;
116	        self.retire_stale_orders(now);
117	        self.persist()?;
118	        Ok(())
119	    }
120	
121	    /// Restore the durable state on the first cycle after a start or restart.
122	    fn ensure_loaded(&mut self) -> Result<(), PortError> {
123	        if self.loaded {
124	            return Ok(());
125	        }
126	        if let Some(snapshot) = self.store.load()? {
127	            self.positions = PositionManager::from_positions(snapshot.positions);
128	            self.pending = PendingOrderManager::from_orders(snapshot.pending_orders);
129	            self.attempted = snapshot.attempted_strategies.into_iter().collect();
130	            self.client_seq = snapshot.client_seq;
131	            self.ledger = AttributionLedger::restore(
132	                snapshot.submissions,
133	                snapshot.absorbed_blocks,
134	                snapshot.fill_cursor,
135	            );
136	            self.closing = snapshot.closing_orders.into_iter().collect();
137	            self.replaced = snapshot.replaced_orders.into_iter().collect();
138	            self.neutralizations = snapshot.neutralizations;
139	        }
140	        self.loaded = true;
141	        Ok(())
142	    }
143	
144	    /// Read both venue sources into the ledger before anything is decided.
145	    ///
146	    /// A failed `open_orders` says nothing about the venue, so no order can be
147	    /// assumed live from it.
148	    fn sync_observations(&mut self) {
149	        match self.gateway.open_orders() {
150	            Ok(open) => {
151	                self.ledger.absorb_open_orders(&open);
152	                self.resting_at_venue = open.into_iter().map(|order| order.order_id).collect();
153	            }
154	            Err(_) => {
155	                self.resting_at_venue.clear();
156	            }
157	        }
158	        if let Ok(batch) = self.fills.poll(&self.ledger.cursor()) {
159	            self.observed_fill_events += self.ledger.absorb_batch(batch);
160	        }
161	    }
162	
163	    /// Match up corrective orders whose fate was unknown when they were sent.
164	    ///
165	    /// A record without an `order_id` is one the venue never acknowledged. The
166	    /// client reference is the venue's idempotency key, so resubmitting under it
167	    /// returns the original order rather than creating a second one.
168	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
169	        let mut resolved: Vec<(usize, OrderId)> = Vec::new();
170	        for (index, record) in self.neutralizations.iter().enumerate() {
171	            if record.order_id.is_some() {
172	                continue;
173	            }
174	            if let Some(order_id) = self.ledger.order_for(&record.client_ref) {
175	                resolved.push((index, order_id));
176	            }
177	        }
178	        for (index, order_id) in resolved {
179	            self.neutralizations[index].order_id = Some(order_id);
180	        }
181	
182	        let unconfirmed: Vec<NeutralizationRecord> = self
183	            .neutralizations
184	            .iter()
185	            .filter(|record| record.order_id.is_none())
186	            .cloned()
187	            .collect();
188	        for record in unconfirmed {
189	            let request = Self::neutralization_request(&record);
190	            if let Ok(accepted) = self.gateway.submit(&request) {
191	                self.ledger.track(
192	                    record.client_ref.clone(),
193	                    Some(accepted.order_id.clone()),
194	                    Some(accepted.venue_seq),
195	                    record.size,
196	                    now,
197	                );
198	                if let Some(slot) = self
199	                    .neutralizations
200	                    .iter_mut()
201	                    .find(|slot| slot.client_ref == record.client_ref)
202	                {
203	                    slot.order_id = Some(accepted.order_id);
204	                }
205	            }
206	        }
207	    }
208	
209	    fn neutralization_request(record: &NeutralizationRecord) -> OrderRequest {
210	        OrderRequest {
211	            strategy_id: record.strategy_id.clone(),
212	            instrument_id: record.instrument_id.clone(),
213	            venue: record.venue,
214	            kind: OrderKind::ImmediateOrCancel,
215	            limit_price: record.limit_price,
216	            size: record.size,
217	            client_ref: record.client_ref.clone(),
218	            expiry_time: record.expiry_time,
219	        }
220	    }
221	
222	    fn enter_new_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
223	        let available_cash = self.inventory.settlement_balance()?;
224	        let candidates = self.strategies.candidates()?;
225	        for candidate in candidates {
226	            if self.positions.has(&candidate.strategy_id) {
227	                continue;
228	            }
229	            if self.attempted.contains(&candidate.strategy_id) {
230	                continue;
231	            }
232	            if candidate.pair_cost() > self.config.max_pair_cost {
233	                continue;
234	            }
235	            if now >= candidate.expiry_time {
236	                continue;
237	            }
238	            let north_size = leg_size(candidate.north_depth, self.config.max_leg_size);
239	            let south_size = leg_size(candidate.south_depth, self.config.max_leg_size);
240	            if north_size < self.config.min_leg_size || south_size < self.config.min_leg_size {
241	                continue;
242	            }
243	            let notional = (candidate.north_price * north_size as f64)
244	                + (candidate.south_price * south_size as f64);
245	            if notional > available_cash {
246	                continue;
247	            }
248	            self.attempted.insert(candidate.strategy_id.clone());
249	            self.place_leg(&candidate, Venue::North, north_size, now);
250	            self.place_leg(&candidate, Venue::South, south_size, now);
251	        }
252	        Ok(())
253	    }
254	
255	    fn place_leg(&mut self, candidate: &StrategyCandidate, venue: Venue, size: Qty, now: Timestamp) {
256	        let client_ref = self.next_client_ref();
257	        let request = OrderRequest {
258	            strategy_id: candidate.strategy_id.clone(),
259	            instrument_id: candidate.instrument_for(venue).clone(),
260	            venue,
261	            kind: OrderKind::RestingLimit,
262	            limit_price: candidate.price_for(venue),
263	            size,
264	            client_ref: client_ref.clone(),
265	            expiry_time: candidate.expiry_time,
266	        };
267	        if let Ok(accepted) = self.gateway.submit(&request) {
268	            self.resting_at_venue.insert(accepted.order_id.clone());
269	            self.ledger.track(
270	                client_ref.clone(),
271	                Some(accepted.order_id.clone()),
272	                Some(accepted.venue_seq),
273	                size,
274	                now,
275	            );
276	            self.pending.insert(PendingOrder::new(
277	                candidate.strategy_id.clone(),
278	                candidate.instrument_for(venue).clone(),
279	                venue,
280	                accepted.order_id,
281	                client_ref,
282	                accepted.venue_seq,
283	                request.limit_price,
284	                size,
285	                candidate.expiry_time,
286	                now,
287	            ));
288	        }
289	    }
290	
291	    /// Quantity attributable to this leg: the resting order itself, plus every
292	    /// corrective order this engine sent for the same mandate and venue.
293	    fn acquired_for(&self, leg: &PendingOrder) -> Qty {
294	        let mut total = self.ledger.attributed_for(&leg.client_ref);
295	        for record in self.records_for(leg) {
296	            total += self.ledger.attributed_for(&record.client_ref);
297	        }
298	        total
299	    }
300	
301	    fn records_for(&self, leg: &PendingOrder) -> Vec<&NeutralizationRecord> {
302	        self.neutralizations
303	            .iter()
304	            .filter(|record| record.strategy_id == leg.strategy_id && record.venue == leg.venue)
305	            .collect()
306	    }
307	
308	    /// Exact integer comparison. There is no tolerance: a leg short by one base
309	    /// unit is short.
310	    fn leg_is_complete(&self, leg: &PendingOrder) -> bool {
311	        self.acquired_for(leg) >= leg.size
312	    }
313	
314	    fn visibility(&self, leg: &PendingOrder) -> LegVisibility {
315	        if self.resting_at_venue.contains(&leg.order_id) {
316	            LegVisibility::Resting
317	        } else if self.replaced.contains(&leg.order_id) {
318	            LegVisibility::Withdrawn
319	        } else {
320	            LegVisibility::Unconfirmed
321	        }
322	    }
323	
324	    fn reconcile_open_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
325	        for strategy_id in self.pending.strategy_ids() {
326	            if self.positions.has(&strategy_id) {
327	                continue;
328	            }
329	            let north_leg = match self.pending.leg(&strategy_id, Venue::North) {
330	                Some(order) => order.clone(),
331	                None => continue,
332	            };
333	            let south_leg = match self.pending.leg(&strategy_id, Venue::South) {
334	                Some(order) => order.clone(),
335	                None => continue,
336	            };
337	
338	            let north_complete = self.leg_is_complete(&north_leg);
339	            let south_complete = self.leg_is_complete(&south_leg);
340	
341	            if north_complete && south_complete {
342	                self.open_position(&north_leg, &south_leg, now);
343	            } else if north_complete {
344	                self.close_gap(&south_leg, now)?;
345	            } else if south_complete {
346	                self.close_gap(&north_leg, now)?;
347	            }
348	        }
349	        Ok(())
350	    }
351	
352	    /// Close the exposure left by a leg that did not complete.
353	    ///
354	    /// Closing a resting leg is a two-step move: the first cycle marks it, the
355	    /// next one cancels it and covers what is missing. That gives an in-flight
356	    /// fill a cycle to arrive before another economic order is sent.
357	    fn close_gap(&mut self, leg: &PendingOrder, now: Timestamp) -> Result<(), PortError> {
358	        match self.visibility(leg) {
359	            LegVisibility::Resting => {
360	                if self.closing.insert(leg.order_id.clone()) {
361	                    return Ok(());
362	                }
363	                if self.gateway.cancel(&leg.order_id).is_err() {
364	                    return Ok(());
365	                }
366	                self.resting_at_venue.remove(&leg.order_id);
367	                self.replaced.insert(leg.order_id.clone());
368	            }
369	            // Not resting at the venue, so there is nothing to cancel first.
370	            LegVisibility::Withdrawn | LegVisibility::Unconfirmed => {}
371	        }
372	
373	        let gap = leg.size - self.acquired_for(leg);
374	        let plan = match plan_neutralization(self.gateway, leg, gap, self.config.neutralization_slippage)? {
375	            Some(plan) => plan,
376	            None => return Ok(()),
377	        };
378	
379	        // The record is written before the order is sent, so a submission whose
380	        // outcome is unknown is still something this engine knows it committed to.
381	        let client_ref = self.next_client_ref();
382	        let record = NeutralizationRecord {
383	            strategy_id: plan.strategy_id,
384	            venue: plan.venue,
385	            instrument_id: plan.instrument_id,
386	            client_ref: client_ref.clone(),
387	            order_id: None,
388	            limit_price: plan.limit_price,
389	            size: plan.size,
390	            expiry_time: leg.expiry_time,
391	        };
392	        let size = record.size;
393	        let request = Self::neutralization_request(&record);
394	        self.neutralizations.push(record);
395	
396	        match self.gateway.submit(&request) {
397	            Ok(accepted) => {
398	                self.ledger.track(
399	                    client_ref.clone(),
400	                    Some(accepted.order_id.clone()),
401	                    Some(accepted.venue_seq),
402	                    size,
403	                    now,
404	                );
405	                if let Some(slot) = self
406	                    .neutralizations
407	                    .iter_mut()
408	                    .find(|slot| slot.client_ref == client_ref)
409	                {
410	                    slot.order_id = Some(accepted.order_id);
411	                }
412	            }
413	            // Nothing came back from the venue, so drop the record and let a later
414	            // cycle work out what is still missing.
415	            Err(_) => {
416	                self.neutralizations.retain(|slot| slot.client_ref != client_ref);
417	            }
418	        }
419	
420	        Ok(())
421	    }
422	
423	    fn open_position(&mut self, north_leg: &PendingOrder, south_leg: &PendingOrder, now: Timestamp) {
424	        let position = Position::new(
425	            north_leg.strategy_id.clone(),
426	            north_leg.size,
427	            south_leg.size,
428	            north_leg.limit_price,
429	            south_leg.limit_price,
430	            north_leg.expiry_time,
431	            now,
432	        );
433	        let strategy_id = position.strategy_id.clone();
434	        self.positions.insert(position);
435	        self.pending.remove(&north_leg.order_id);
436	        self.pending.remove(&south_leg.order_id);
437	        self.neutralizations
438	            .retain(|record| record.strategy_id != strategy_id);
439	    }
440	
441	    fn retire_stale_orders(&mut self, now: Timestamp) {
442	        for order_id in self.pending.expiring(now) {
443	            let _ = self.gateway.cancel(&order_id);
444	            self.pending.remove(&order_id);
445	            self.closing.remove(&order_id);
446	            self.replaced.remove(&order_id);
447	        }
448	    }
449	
450	    fn persist(&self) -> Result<(), PortError> {
451	        let snapshot = Snapshot {
452	            positions: self.positions.all(),
453	            pending_orders: self.pending.all(),
454	            attempted_strategies: self.attempted.iter().cloned().collect(),
455	            client_seq: self.client_seq,
456	            fill_cursor: self.ledger.cursor(),
457	            submissions: self.ledger.submissions(),
458	            absorbed_blocks: self.ledger.absorbed_blocks(),
459	            closing_orders: self.closing.iter().cloned().collect(),
460	            replaced_orders: self.replaced.iter().cloned().collect(),
461	            neutralizations: self.neutralizations.clone(),
462	        };
463	        self.store.save(&snapshot)
464	    }
465	
466	    fn next_client_ref(&mut self) -> ClientRef {
467	        self.client_seq += 1;
468	        ClientRef::new(format!(
469	            "{}-{}",
470	            self.config.client_ref_prefix, self.client_seq
471	        ))
472	    }
