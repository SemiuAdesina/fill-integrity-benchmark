2	//! One engine cycle: read the venue, decide what completed, act, persist.
3	//!
4	//! `run_cycle` is the only entry point the graded suite calls. The order of the
5	//! steps below is the order they run in, and everything else in the crate is a
6	//! type or a helper reached from here.
7

26	/// What the engine currently knows about whether a leg is still working.
27	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28	enum LegVisibility {
29	    /// Present in `open_orders`.
30	    Resting,
31	    /// Absent from `open_orders`, and this engine cancelled it.
32	    Withdrawn,
33	    /// Absent from `open_orders`, and the venue has closed it: a block naming
34	    /// this order has been observed, so it will not acquire another base unit.
35	    Closed,
36	    /// Absent from `open_orders`, and neither of the above is known. It may
37	    /// still be resting and may still fill.
38	    Unconfirmed,
39	}
40	
41	pub struct Engine<'a> {
42	    config: EngineConfig,
43	    strategies: &'a dyn StrategySource,
44	    inventory: &'a dyn InventorySource,
45	    gateway: &'a dyn OrderGateway,
46	    fills: &'a dyn FillEventSource,
47	    store: &'a dyn StateStore,
48	    clock: &'a dyn Clock,
49	    pending: PendingOrderManager,
50	    positions: PositionManager,
51	    attempted: BTreeSet<StrategyId>,
52	    /// Per-order attribution state, and the feed cursor that produced it.
53	    ledger: AttributionLedger,
54	    /// Corrective orders this engine has committed to, resolved or not.
55	    neutralizations: Vec<NeutralizationRecord>,
56	    /// Legs this engine has moved into closure at the venue.
57	    closing: BTreeSet<OrderId>,
58	    /// Legs this engine has cancelled at the venue.
59	    replaced: BTreeSet<OrderId>,
60	    /// Order ids the last `open_orders` call reported as live. Not durable: it is
61	    /// re-read every cycle and says nothing after a restart.
62	    resting_at_venue: BTreeSet<OrderId>,
63	    loaded: bool,
64	    client_seq: u64,
65	    observed_fill_events: usize,
66	}

113	    pub fn run_cycle(&mut self) -> Result<(), PortError> {
114	        self.ensure_loaded()?;
115	        let now = self.clock.now();
116	        self.sync_observations();
117	        self.reconcile_neutralizations(now);
118	        self.enter_new_strategies(now)?;
119	        self.reconcile_open_strategies(now)?;
120	        self.retire_stale_orders(now);
121	        self.persist()?;
122	        Ok(())
123	    }
124	
125	    /// Restore the durable state on the first cycle after a start or restart.
126	    ///
127	    /// A restart drops everything in memory but nothing at the venue or in
128	    /// custody, so what comes back has to include the original attribution
129	    /// window starts, the feed position, the attribution already accumulated,
130	    /// the absorbed block references, the work still in progress and the
131	    /// `client_ref` identity a retry depends on.
132	    fn ensure_loaded(&mut self) -> Result<(), PortError> {
133	        if self.loaded {
134	            return Ok(());
135	        }
136	        if let Some(snapshot) = self.store.load()? {
137	            self.positions = PositionManager::from_positions(snapshot.positions);
138	            self.pending = PendingOrderManager::from_orders(snapshot.pending_orders);
139	            self.attempted = snapshot.attempted_strategies.into_iter().collect();
140	            self.client_seq = snapshot.client_seq;
141	            self.ledger = AttributionLedger::restore(
142	                snapshot.submissions,
143	                snapshot.absorbed_blocks,
144	                snapshot.fill_cursor,
145	            );
146	            self.closing = snapshot.closing_orders.into_iter().collect();
147	            self.replaced = snapshot.replaced_orders.into_iter().collect();
148	            self.neutralizations = snapshot.neutralizations;
149	        }
150	        self.loaded = true;
151	        Ok(())
152	    }
153	
154	    /// Read both venue sources into the ledger before anything is decided.
155	    ///
156	    /// A failed `open_orders` says nothing about the venue, so no order can be
157	    /// assumed live from it.
158	    fn sync_observations(&mut self) {
159	        match self.gateway.open_orders() {
160	            Ok(open) => {
161	                self.ledger.absorb_open_orders(&open);
162	                self.resting_at_venue = open.into_iter().map(|order| order.order_id).collect();
163	            }
164	            Err(_) => {
165	                self.resting_at_venue.clear();
166	            }
167	        }
168	        if let Ok(batch) = self.fills.poll(&self.ledger.cursor()) {
169	            self.observed_fill_events += self.ledger.absorb_batch(batch);
170	        }
171	    }
172	
173	    /// Match up corrective orders whose fate was unknown when they were sent.
174	    ///
175	    /// A record without an `order_id` is one the venue never acknowledged. The
176	    /// venue may have created it, so it stays committed and is retried — under
177	    /// the *same* `client_ref`, which is the venue's idempotency key, so the
178	    /// retry resolves onto the original order instead of placing a second one.
179	    /// Identity is also recovered passively, when a later observation carries
180	    /// that `client_ref`.
181	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
182	        let mut resolved: Vec<(usize, OrderId)> = Vec::new();
183	        for (index, record) in self.neutralizations.iter().enumerate() {
184	            if record.order_id.is_some() {
185	                continue;
186	            }
187	            if let Some(order_id) = self.ledger.order_for(&record.client_ref) {
188	                resolved.push((index, order_id));
189	            }
190	        }
191	        for (index, order_id) in resolved {
192	            self.neutralizations[index].order_id = Some(order_id);
193	        }
194	
195	        // A retry only makes sense while the leg it was meant to cover is still
196	        // being worked: once the leg is gone there is no exposure to resolve.
197	        let unconfirmed: Vec<NeutralizationRecord> = self
198	            .neutralizations
199	            .iter()
200	            .filter(|record| record.order_id.is_none())
201	            .filter(|record| self.pending.leg(&record.strategy_id, record.venue).is_some())
202	            .cloned()
203	            .collect();
204	        for record in unconfirmed {
205	            let request = Self::neutralization_request(&record);
206	            match self.gateway.submit(&request) {
207	                Ok(accepted) => {
208	                    self.ledger.track(
209	                        record.client_ref.clone(),
210	                        Some(accepted.order_id.clone()),
211	                        Some(accepted.venue_seq),
212	                        record.size,
213	                        now,
214	                    );
215	                    if let Some(slot) = self
216	                        .neutralizations
217	                        .iter_mut()
218	                        .find(|slot| slot.client_ref == record.client_ref)
219	                    {
220	                        slot.order_id = Some(accepted.order_id);
221	                    }
222	                }
223	                // Still unresolved, so it stays committed and is retried again
224	                // on a later cycle under this same reference.
225	                Err(SubmitError::AcknowledgementLost) => {}
226	                // The venue has answered for that reference, so nothing exists
227	                // under it: nothing is working, and the record goes. A later
228	                // cycle recomputes the residual from scratch and tries again
229	                // under a fresh reference.
230	                Err(SubmitError::Rejected(_)) | Err(SubmitError::Transport(_)) => {
231	                    self.neutralizations
232	                        .retain(|slot| slot.client_ref != record.client_ref);
233	                }
234	            }
235	        }
236	    }
237	
238	    fn neutralization_request(record: &NeutralizationRecord) -> OrderRequest {
239	        OrderRequest {
240	            strategy_id: record.strategy_id.clone(),
241	            instrument_id: record.instrument_id.clone(),
242	            venue: record.venue,
243	            kind: OrderKind::ImmediateOrCancel,
244	            limit_price: record.limit_price,
245	            size: record.size,
246	            client_ref: record.client_ref.clone(),
247	            expiry_time: record.expiry_time,
248	        }
249	    }
250	
251	    fn enter_new_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
252	        let available_cash = self.inventory.settlement_balance()?;
253	        let candidates = self.strategies.candidates()?;
254	        for candidate in candidates {
255	            if self.positions.has(&candidate.strategy_id) {
256	                continue;
257	            }
258	            if self.attempted.contains(&candidate.strategy_id) {
259	                continue;
260	            }
261	            if candidate.pair_cost() > self.config.max_pair_cost {
262	                continue;
263	            }
264	            if now >= candidate.expiry_time {
265	                continue;
266	            }
267	            let north_size = leg_size(candidate.north_depth, self.config.max_leg_size);
268	            let south_size = leg_size(candidate.south_depth, self.config.max_leg_size);
269	            if north_size < self.config.min_leg_size || south_size < self.config.min_leg_size {
270	                continue;
271	            }
272	            let notional = (candidate.north_price * north_size as f64)
273	                + (candidate.south_price * south_size as f64);
274	            if notional > available_cash {
275	                continue;
276	            }
277	            self.attempted.insert(candidate.strategy_id.clone());
278	            self.place_leg(&candidate, Venue::North, north_size, now);
279	            self.place_leg(&candidate, Venue::South, south_size, now);
280	        }
281	        Ok(())
282	    }
283	
284	    fn place_leg(&mut self, candidate: &StrategyCandidate, venue: Venue, size: Qty, now: Timestamp) {
285	        let client_ref = self.next_client_ref();
286	        let request = OrderRequest {
287	            strategy_id: candidate.strategy_id.clone(),
288	            instrument_id: candidate.instrument_for(venue).clone(),
289	            venue,
290	            kind: OrderKind::RestingLimit,
291	            limit_price: candidate.price_for(venue),
292	            size,
293	            client_ref: client_ref.clone(),
294	            expiry_time: candidate.expiry_time,
295	        };
296	        if let Ok(accepted) = self.gateway.submit(&request) {
297	            self.resting_at_venue.insert(accepted.order_id.clone());
298	            self.ledger.track(
299	                client_ref.clone(),
300	                Some(accepted.order_id.clone()),
301	                Some(accepted.venue_seq),
302	                size,
303	                now,
304	            );
305	            self.pending.insert(PendingOrder::new(
306	                candidate.strategy_id.clone(),
307	                candidate.instrument_for(venue).clone(),
308	                venue,
309	                accepted.order_id,
310	                client_ref,
311	                accepted.venue_seq,
312	                request.limit_price,
313	                size,
314	                candidate.expiry_time,
315	                now,
316	            ));
317	        }
318	    }

320	    /// Quantity attributable to this leg: the resting order itself, plus every
321	    /// corrective order this engine sent for the same mandate and venue.
322	    ///
323	    /// A neutralization fill resolves the exposure; it must not also be counted
324	    /// a second time as a fill of the original resting order, so the two are
325	    /// added here rather than compared.
326	    fn acquired_for(&self, leg: &PendingOrder) -> Qty {
327	        let mut total = self.ledger.attributed_for(&leg.client_ref);
328	        for record in self.records_for(leg) {
329	            total += self.ledger.attributed_for(&record.client_ref);
330	        }
331	        total
332	    }
333	
334	    fn records_for(&self, leg: &PendingOrder) -> Vec<&NeutralizationRecord> {
335	        self.neutralizations
336	            .iter()
337	            .filter(|record| record.strategy_id == leg.strategy_id && record.venue == leg.venue)
338	            .collect()
339	    }
340	
341	    /// Quantity still working in a committed neutralization.
342	    ///
343	    /// This is what stops consecutive cycles from sending the same exposure
344	    /// twice. A neutralization the engine has committed to counts as working
345	    /// until it resolves:
346	    ///
347	    /// - one the venue acknowledged contributes its own outstanding quantity;
348	    /// - one whose acknowledgement was lost contributes its full size, because
349	    ///   the venue may have created it and the engine cannot yet see it;
350	    /// - one the venue has closed contributes nothing, whatever it managed to
351	    ///   fill, because the gap it leaves behind is real.
352	    fn working_in(&self, record: &NeutralizationRecord) -> Qty {
353	        if self.ledger.is_finalized(&record.client_ref) {
354	            return 0;
355	        }
356	        if record.order_id.is_none() {
357	            return record.size;
358	        }
359	        let outstanding = record.size - self.ledger.attributed_for(&record.client_ref);
360	        if outstanding > 0 {
361	            outstanding
362	        } else {
363	            0
364	        }
365	    }
366	
367	    /// The exposure still to cover on a leg: the required quantity, less what
368	    /// is already attributed to it, less what is still working in
369	    /// neutralizations this engine has already committed to.
370	    fn residual_for(&self, leg: &PendingOrder) -> Qty {
371	        let mut residual = leg.size - self.acquired_for(leg);
372	        for record in self.records_for(leg) {
373	            residual -= self.working_in(record);
374	        }
375	        residual
376	    }
377	
378	    /// Exact integer comparison. There is no tolerance: a leg short by one base
379	    /// unit is short.
380	    fn leg_is_complete(&self, leg: &PendingOrder) -> bool {
381	        self.acquired_for(leg) >= leg.size
382	    }
383	
384	    fn visibility(&self, leg: &PendingOrder) -> LegVisibility {
385	        if self.resting_at_venue.contains(&leg.order_id) {
386	            LegVisibility::Resting
387	        } else if self.replaced.contains(&leg.order_id) {
388	            LegVisibility::Withdrawn
389	        } else if self.ledger.is_finalized(&leg.client_ref) {
390	            // Positive evidence from the venue rather than an absence to guess
391	            // about: a block named this order, so it is closed and cancelled
392	            // and will not fill again.
393	            LegVisibility::Closed
394	        } else {
395	            LegVisibility::Unconfirmed
396	        }
397	    }
398	
399	    fn reconcile_open_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
400	        for strategy_id in self.pending.strategy_ids() {
401	            if self.positions.has(&strategy_id) {
402	                continue;
403	            }
404	            let north_leg = match self.pending.leg(&strategy_id, Venue::North) {
405	                Some(order) => order.clone(),
406	                None => continue,
407	            };
408	            let south_leg = match self.pending.leg(&strategy_id, Venue::South) {
409	                Some(order) => order.clone(),
410	                None => continue,
411	            };
412	
413	            let north_complete = self.leg_is_complete(&north_leg);
414	            let south_complete = self.leg_is_complete(&south_leg);
415	
416	            if north_complete && south_complete {
417	                self.open_position(&north_leg, &south_leg, now);
418	            } else if north_complete {
419	                self.close_gap(&south_leg, now)?;
420	            } else if south_complete {
421	                self.close_gap(&north_leg, now)?;
422	            }
423	        }
424	        Ok(())
425	    }
426	
427	    /// Close the exposure left by a leg that did not complete.
428	    ///
429	    /// Correcting a resting leg is a two-step move. The cycle that first finds
430	    /// the exposure moves the resting order into closure, cancelling it at the
431	    /// venue; only a later cycle, once that cancellation boundary has passed,
432	    /// issues the corrective order for whatever residual then remains. The
433	    /// ordering exists so that a fill already in flight has a cycle in which to
434	    /// land and be attributed before another economic order is sent.
435	    ///
436	    /// An order the venue never confirmed is never neutralized against: it may
437	    /// still be resting and may still fill, and neutralizing quantity the desk
438	    /// already owns doubles the exposure the correction was meant to close.
439	    fn close_gap(&mut self, leg: &PendingOrder, now: Timestamp) -> Result<(), PortError> {
440	        match self.visibility(leg) {
441	            LegVisibility::Resting => {
442	                if !self.closing.contains(&leg.order_id) {
443	                    // First cycle for this leg: move it into closure and stop
444	                    // there. The corrective order waits for the next cycle.
445	                    if self.gateway.cancel(&leg.order_id).is_err() {
446	                        return Ok(());
447	                    }
448	                    self.closing.insert(leg.order_id.clone());
449	                    self.replaced.insert(leg.order_id.clone());
450	                    self.resting_at_venue.remove(&leg.order_id);
451	                    return Ok(());
452	                }
453	                // Already moved into closure on an earlier cycle. The venue may
454	                // still be listing it, but this engine has cancelled it, so it
455	                // is no longer working and the residual below is real.
456	            }
457	            // Absent from the live list and not withdrawn by this engine: its
458	            // state is unknown, so nothing is concluded and nothing is sent.
459	            LegVisibility::Unconfirmed => return Ok(()),
460	            // Withdrawn by this engine or closed by the venue: nothing is
461	            // resting, so the residual can be covered.
462	            LegVisibility::Withdrawn | LegVisibility::Closed => {}
463	        }
464	
465	        let residual = self.residual_for(leg);
466	        let plan = match plan_neutralization(
467	            self.gateway,
468	            leg,
469	            residual,
470	            self.config.neutralization_slippage,
471	        )? {
472	            Some(plan) => plan,
473	            None => return Ok(()),
474	        };
475	
476	        // The record is written before the order is sent, so a submission whose
477	        // outcome is unknown is still something this engine knows it committed
478	        // to. The submission is tracked under its `client_ref` for the same
479	        // reason: it is the key a later observation carries back.
480	        let client_ref = self.next_client_ref();
481	        let record = NeutralizationRecord {
482	            strategy_id: plan.strategy_id,
483	            venue: plan.venue,
484	            instrument_id: plan.instrument_id,
485	            client_ref: client_ref.clone(),
486	            order_id: None,
487	            limit_price: plan.limit_price,
488	            size: plan.size,
489	            expiry_time: leg.expiry_time,
490	        };
491	        let size = record.size;
492	        let request = Self::neutralization_request(&record);
493	        self.ledger
494	            .track(client_ref.clone(), None, None, size, now);
495	        self.neutralizations.push(record);
496	
497	        match self.gateway.submit(&request) {
498	            Ok(accepted) => {
499	                self.ledger.track(
500	                    client_ref.clone(),
501	                    Some(accepted.order_id.clone()),
502	                    Some(accepted.venue_seq),
503	                    size,
504	                    now,
505	                );
506	                if let Some(slot) = self
507	                    .neutralizations
508	                    .iter_mut()
509	                    .find(|slot| slot.client_ref == client_ref)
510	                {
511	                    slot.order_id = Some(accepted.order_id);
512	                }
513	            }
514	            // The venue may have created the order, so it stays committed: the
515	            // record keeps its `client_ref`, contributes its full size as
516	            // working, and is retried under that same reference.
517	            Err(SubmitError::AcknowledgementLost) => {}
518	            // Nothing was created, so nothing is working. Drop the record and
519	            // let a later cycle recompute the residual and try again: a mandate
520	            // is not latched shut by one failed attempt.
521	            Err(SubmitError::Rejected(_)) | Err(SubmitError::Transport(_)) => {
522	                self.neutralizations
523	                    .retain(|slot| slot.client_ref != client_ref);
524	            }
525	        }
526	
527	        Ok(())
528	    }

530	    fn open_position(&mut self, north_leg: &PendingOrder, south_leg: &PendingOrder, now: Timestamp) {
531	        let position = Position::new(
532	            north_leg.strategy_id.clone(),
533	            north_leg.size,
534	            south_leg.size,
535	            north_leg.limit_price,
536	            south_leg.limit_price,
537	            north_leg.expiry_time,
538	            now,
539	        );
540	        let strategy_id = position.strategy_id.clone();
541	        self.positions.insert(position);
542	        self.pending.remove(&north_leg.order_id);
543	        self.pending.remove(&south_leg.order_id);
544	        self.neutralizations
545	            .retain(|record| record.strategy_id != strategy_id);
546	    }
547	
548	    fn retire_stale_orders(&mut self, now: Timestamp) {
549	        for order_id in self.pending.expiring(now) {
550	            let _ = self.gateway.cancel(&order_id);
551	            self.pending.remove(&order_id);
552	            self.closing.remove(&order_id);
553	            self.replaced.remove(&order_id);
554	        }
555	    }
556	
557	    fn persist(&self) -> Result<(), PortError> {
558	        let snapshot = Snapshot {
559	            positions: self.positions.all(),
560	            pending_orders: self.pending.all(),
561	            attempted_strategies: self.attempted.iter().cloned().collect(),
562	            client_seq: self.client_seq,
563	            fill_cursor: self.ledger.cursor(),
564	            submissions: self.ledger.submissions(),
565	            absorbed_blocks: self.ledger.absorbed_blocks(),
566	            closing_orders: self.closing.iter().cloned().collect(),
567	            replaced_orders: self.replaced.iter().cloned().collect(),
568	            neutralizations: self.neutralizations.clone(),
569	        };
570	        self.store.save(&snapshot)
571	    }
572	
573	    fn next_client_ref(&mut self) -> ClientRef {
574	        self.client_seq += 1;
575	        ClientRef::new(format!(
576	            "{}-{}",
577	            self.config.client_ref_prefix, self.client_seq
578	        ))
579	    }
