181	    /// Resolve an observation to one of our submissions.
182	    ///
183	    /// The `client_ref` the engine chose is the primary key: it is the key the
184	    /// engine always knows, and it is the only key that stays unambiguous when
185	    /// the venue recycles an `order_id`. The venue's label is the fallback, and
186	    /// it is how an observation reaches a submission the venue never
187	    /// acknowledged — learning the id this way is what resolves a lost
188	    /// acknowledgement later.
189	    ///
190	    /// A timestamped observation only resolves by `order_id` to a submission
191	    /// whose own attribution window contains it. Outside that window the label
192	    /// names some other order, and counting the observation is a
193	    /// misattribution rather than a rounding matter.
194	    fn resolve(
195	        &mut self,
196	        client_ref: &ClientRef,
197	        order_id: &OrderId,
198	        at: Option<Timestamp>,
199	    ) -> Option<usize> {
200	        if let Some(index) = self.index_by_client_ref.get(client_ref).copied() {
201	            if self.submissions[index].order_id.is_none() {
202	                self.submissions[index].order_id = Some(order_id.clone());
203	            }
204	            return Some(index);
205	        }
206	        self.locate_by_order(order_id, at)
207	    }

209	    /// The submission a venue label most plausibly names right now.
210	    fn locate_by_order(&self, order_id: &OrderId, at: Option<Timestamp>) -> Option<usize> {
211	        let mut candidates = self.indices_of_order(order_id);
212	        if let Some(at) = at {
213	            candidates.retain(|index| Self::in_window(&self.submissions[*index], at));
214	        }
215	        candidates.into_iter().max_by_key(|index| {
216	            let record = &self.submissions[*index];
217	            (record.window_start, *index)
218	        })
219	    }

221	    /// Whether a venue match timestamp falls in this record's attribution window.
222	    ///
223	    /// The window is half-open, `[start, start + ATTRIBUTION_WINDOW)`: an
224	    /// observation exactly at the start belongs to the order, one exactly at the
225	    /// end does not. It is the venue's own match timestamp that is tested, never
226	    /// the cycle in which the engine happened to read the observation.
227	    fn in_window(record: &SubmissionRecord, at: Timestamp) -> bool {
228	        at >= record.window_start && at < record.window_start + ATTRIBUTION_WINDOW
229	    }

231	    /// Fold one polled batch into the ledger and advance the cursor.
232	    ///
233	    /// Returns the number of per-order events in the batch, which the engine
234	    /// reports through `Engine::observed_fill_events`.
235	    pub fn absorb_batch(&mut self, batch: FillBatch) -> usize {
236	        let count = batch.events.len();
237	        for event in batch.events {
238	            self.absorb_event(event);
239	        }
240	        for block in batch.blocks {
241	            self.absorb_block(block);
242	        }
243	        self.cursor = batch.next_cursor;
244	        count
245	    }

247	    /// Fold one per-order fill event in.
248	    ///
249	    /// `cumulative_filled_qty` is the total that order has filled so far, not
250	    /// the size of this particular match, and the feed may redeliver it, replay
251	    /// it long afterwards or deliver it out of order. It is therefore a
252	    /// high-water mark: repeated, replayed or stale readings move the mark and
253	    /// never add quantity.
254	    fn absorb_event(&mut self, event: FillEvent) {
255	        let Some(index) = self.resolve(&event.client_ref, &event.order_id, Some(event.at)) else {
256	            return;
257	        };
258	        if !Self::in_window(&self.submissions[index], event.at) {
259	            return;
260	        }
261	        let record = &mut self.submissions[index];
262	        if event.cumulative_filled_qty > record.direct_filled {
263	            record.direct_filled = event.cumulative_filled_qty;
264	        }
265	    }
