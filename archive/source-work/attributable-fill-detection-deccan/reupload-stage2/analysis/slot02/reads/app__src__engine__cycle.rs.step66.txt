2	//! One engine cycle: read the venue, decide what completed, act, persist.
3	//!
4	//! `run_cycle` is the only entry point the graded suite calls. The order of the
5	//! steps below is the order they run in, and everything else in the crate is a
6	//! type or a helper reached from here.
7

26	/// What the engine currently knows about whether a leg is still working.
27	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28	enum LegVisibility {
29	    /// Present in `open_orders`.
30	    Resting,
31	    /// Absent from `open_orders`, and this engine cancelled it.
32	    Withdrawn,
33	    /// Absent from `open_orders`, and the venue has closed it: a block naming
34	    /// this order has been observed, so it will not acquire another base unit.
35	    Closed,
36	    /// Absent from `open_orders`, and neither of the above is known. It may
37	    /// still be resting and may still fill.
38	    Unconfirmed,
39	}
40	
41	pub struct Engine<'a> {
42	    config: EngineConfig,
43	    strategies: &'a dyn StrategySource,
44	    inventory: &'a dyn InventorySource,
45	    gateway: &'a dyn OrderGateway,
46	    fills: &'a dyn FillEventSource,
47	    store: &'a dyn StateStore,
48	    clock: &'a dyn Clock,
49	    pending: PendingOrderManager,
50	    positions: PositionManager,
51	    attempted: BTreeSet<StrategyId>,
52	    /// Per-order attribution state, and the feed cursor that produced it.
53	    ledger: AttributionLedger,
54	    /// Corrective orders this engine has committed to, resolved or not.
55	    neutralizations: Vec<NeutralizationRecord>,
56	    /// Legs this engine has moved into closure at the venue.
57	    closing: BTreeSet<OrderId>,
58	    /// Legs this engine has cancelled at the venue.
59	    replaced: BTreeSet<OrderId>,
60	    /// Order ids the last `open_orders` call reported as live. Not durable: it is
61	    /// re-read every cycle and says nothing after a restart.
62	    resting_at_venue: BTreeSet<OrderId>,
63	    loaded: bool,
64	    client_seq: u64,
65	    observed_fill_events: usize,
66	}

113	    pub fn run_cycle(&mut self) -> Result<(), PortError> {
114	        self.ensure_loaded()?;
115	        let now = self.clock.now();
116	        self.sync_observations();
117	        self.reconcile_neutralizations(now);
118	        self.enter_new_strategies(now)?;
119	        self.reconcile_open_strategies(now)?;
120	        self.retire_stale_orders(now);
121	        self.persist()?;
122	        Ok(())
123	    }
124	
125	    /// Restore the durable state on the first cycle after a start or restart.
126	    ///
127	    /// A restart drops everything in memory but nothing at the venue or in
128	    /// custody, so what comes back has to include the original attribution
129	    /// window starts, the feed position, the attribution already accumulated,
130	    /// the absorbed block references, the work still in progress and the
131	    /// `client_ref` identity a retry depends on.
132	    fn ensure_loaded(&mut self) -> Result<(), PortError> {
133	        if self.loaded {
134	            return Ok(());
135	        }
136	        if let Some(snapshot) = self.store.load()? {
137	            self.positions = PositionManager::from_positions(snapshot.positions);
138	            self.pending = PendingOrderManager::from_orders(snapshot.pending_orders);
139	            self.attempted = snapshot.attempted_strategies.into_iter().collect();
140	            self.client_seq = snapshot.client_seq;
141	            self.ledger = AttributionLedger::restore(
142	                snapshot.submissions,
143	                snapshot.absorbed_blocks,
144	                snapshot.fill_cursor,
145	            );
146	            self.closing = snapshot.closing_orders.into_iter().collect();
147	            self.replaced = snapshot.replaced_orders.into_iter().collect();
148	            self.neutralizations = snapshot.neutralizations;
149	        }
150	        self.loaded = true;
151	        Ok(())
152	    }
153	
154	    /// Read both venue sources into the ledger before anything is decided.
155	    ///
156	    /// A failed `open_orders` says nothing about the venue, so no order can be
157	    /// assumed live from it.
158	    fn sync_observations(&mut self) {
159	        match self.gateway.open_orders() {
160	            Ok(open) => {
161	                self.ledger.absorb_open_orders(&open);
162	                self.resting_at_venue = open.into_iter().map(|order| order.order_id).collect();
163	            }
164	            Err(_) => {
165	                self.resting_at_venue.clear();
166	            }
167	        }
168	        if let Ok(batch) = self.fills.poll(&self.ledger.cursor()) {
169	            self.observed_fill_events += self.ledger.absorb_batch(batch);
170	        }
171	    }
172	
173	    /// Match up corrective orders whose fate was unknown when they were sent.
174	    ///
175	    /// A record without an `order_id` is one the venue never acknowledged. The
176	    /// venue may have created it, so it stays committed and is retried — under
177	    /// the *same* `client_ref`, which is the venue's idempotency key, so the
178	    /// retry resolves onto the original order instead of placing a second one.
179	    /// Identity is also recovered passively, when a later observation carries
180	    /// that `client_ref`.
181	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
182	        let mut resolved: Vec<(usize, OrderId)> = Vec::new();
183	        for (index, record) in self.neutralizations.iter().enumerate() {
184	            if record.order_id.is_some() {
185	                continue;
186	            }
187	            if let Some(order_id) = self.ledger.order_for(&record.client_ref) {
188	                resolved.push((index, order_id));
189	            }
190	        }
191	        for (index, order_id) in resolved {
192	            self.neutralizations[index].order_id = Some(order_id);
193	        }
194	
195	        let unconfirmed: Vec<NeutralizationRecord> = self
196	            .neutralizations
197	            .iter()
198	            .filter(|record| record.order_id.is_none())
199	            .cloned()
200	            .collect();
201	        for record in unconfirmed {
202	            let request = Self::neutralization_request(&record);
203	            if let Ok(accepted) = self.gateway.submit(&request) {
204	                self.ledger.track(
205	                    record.client_ref.clone(),
206	                    Some(accepted.order_id.clone()),
207	                    Some(accepted.venue_seq),
208	                    record.size,
209	                    now,
210	                );
211	                if let Some(slot) = self
212	                    .neutralizations
213	                    .iter_mut()
214	                    .find(|slot| slot.client_ref == record.client_ref)
215	                {
216	                    slot.order_id = Some(accepted.order_id);
217	                }
218	            }
219	        }
220	    }
221	
222	    fn neutralization_request(record: &NeutralizationRecord) -> OrderRequest {
223	        OrderRequest {
224	            strategy_id: record.strategy_id.clone(),
225	            instrument_id: record.instrument_id.clone(),
226	            venue: record.venue,
227	            kind: OrderKind::ImmediateOrCancel,
228	            limit_price: record.limit_price,
229	            size: record.size,
230	            client_ref: record.client_ref.clone(),
231	            expiry_time: record.expiry_time,
232	        }
233	    }
234	
235	    fn enter_new_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
236	        let available_cash = self.inventory.settlement_balance()?;
237	        let candidates = self.strategies.candidates()?;
238	        for candidate in candidates {
239	            if self.positions.has(&candidate.strategy_id) {
240	                continue;
241	            }
242	            if self.attempted.contains(&candidate.strategy_id) {
243	                continue;
244	            }
245	            if candidate.pair_cost() > self.config.max_pair_cost {
246	                continue;
247	            }
248	            if now >= candidate.expiry_time {
249	                continue;
250	            }
251	            let north_size = leg_size(candidate.north_depth, self.config.max_leg_size);
252	            let south_size = leg_size(candidate.south_depth, self.config.max_leg_size);
253	            if north_size < self.config.min_leg_size || south_size < self.config.min_leg_size {
254	                continue;
255	            }
256	            let notional = (candidate.north_price * north_size as f64)
257	                + (candidate.south_price * south_size as f64);
258	            if notional > available_cash {
259	                continue;
260	            }
261	            self.attempted.insert(candidate.strategy_id.clone());
262	            self.place_leg(&candidate, Venue::North, north_size, now);
263	            self.place_leg(&candidate, Venue::South, south_size, now);
264	        }
265	        Ok(())
266	    }
267	
268	    fn place_leg(&mut self, candidate: &StrategyCandidate, venue: Venue, size: Qty, now: Timestamp) {
269	        let client_ref = self.next_client_ref();
270	        let request = OrderRequest {
271	            strategy_id: candidate.strategy_id.clone(),
272	            instrument_id: candidate.instrument_for(venue).clone(),
273	            venue,
274	            kind: OrderKind::RestingLimit,
275	            limit_price: candidate.price_for(venue),
276	            size,
277	            client_ref: client_ref.clone(),
278	            expiry_time: candidate.expiry_time,
279	        };
280	        if let Ok(accepted) = self.gateway.submit(&request) {
281	            self.resting_at_venue.insert(accepted.order_id.clone());
282	            self.ledger.track(
283	                client_ref.clone(),
284	                Some(accepted.order_id.clone()),
285	                Some(accepted.venue_seq),
286	                size,
287	                now,
288	            );
289	            self.pending.insert(PendingOrder::new(
290	                candidate.strategy_id.clone(),
291	                candidate.instrument_for(venue).clone(),
292	                venue,
293	                accepted.order_id,
294	                client_ref,
295	                accepted.venue_seq,
296	                request.limit_price,
297	                size,
298	                candidate.expiry_time,
299	                now,
300	            ));
301	        }
302	    }
303	
304	    /// Quantity attributable to this leg: the resting order itself, plus every
305	    /// corrective order this engine sent for the same mandate and venue.
306	    ///
307	    /// A neutralization fill resolves the exposure; it must not also be counted
308	    /// a second time as a fill of the original resting order, so the two are
309	    /// added here rather than compared.
310	    fn acquired_for(&self, leg: &PendingOrder) -> Qty {
311	        let mut total = self.ledger.attributed_for(&leg.client_ref);
312	        for record in self.records_for(leg) {
313	            total += self.ledger.attributed_for(&record.client_ref);
314	        }
315	        total
316	    }
317	
318	    fn records_for(&self, leg: &PendingOrder) -> Vec<&NeutralizationRecord> {
319	        self.neutralizations
320	            .iter()
321	            .filter(|record| record.strategy_id == leg.strategy_id && record.venue == leg.venue)
322	            .collect()
323	    }
324	
325	    /// Quantity still working in a committed neutralization.
326	    ///
327	    /// This is what stops consecutive cycles from sending the same exposure
328	    /// twice. A neutralization the engine has committed to counts as working
329	    /// until it resolves:
330	    ///
331	    /// - one the venue acknowledged contributes its own outstanding quantity;
332	    /// - one whose acknowledgement was lost contributes its full size, because
333	    ///   the venue may have created it and the engine cannot yet see it;
334	    /// - one the venue has closed contributes nothing, whatever it managed to
335	    ///   fill, because the gap it leaves behind is real.
336	    fn working_in(&self, record: &NeutralizationRecord) -> Qty {
337	        if self.ledger.is_finalized(&record.client_ref) {
338	            return 0;
339	        }
340	        if record.order_id.is_none() {
341	            return record.size;
342	        }
343	        let outstanding = record.size - self.ledger.attributed_for(&record.client_ref);
344	        if outstanding > 0 {
345	            outstanding
346	        } else {
347	            0
348	        }
349	    }
350	
351	    /// The exposure still to cover on a leg: the required quantity, less what
352	    /// is already attributed to it, less what is still working in
353	    /// neutralizations this engine has already committed to.
354	    fn residual_for(&self, leg: &PendingOrder) -> Qty {
355	        let mut residual = leg.size - self.acquired_for(leg);
356	        for record in self.records_for(leg) {
357	            residual -= self.working_in(record);
358	        }
359	        residual
360	    }
361	
362	    /// Exact integer comparison. There is no tolerance: a leg short by one base
363	    /// unit is short.
364	    fn leg_is_complete(&self, leg: &PendingOrder) -> bool {
365	        self.acquired_for(leg) >= leg.size
366	    }
367	
368	    fn visibility(&self, leg: &PendingOrder) -> LegVisibility {
369	        if self.resting_at_venue.contains(&leg.order_id) {
370	            LegVisibility::Resting
371	        } else if self.replaced.contains(&leg.order_id) {
372	            LegVisibility::Withdrawn
373	        } else if self.ledger.is_finalized(&leg.client_ref) {
374	            // Positive evidence from the venue rather than an absence to guess
375	            // about: a block named this order, so it is closed and cancelled
376	            // and will not fill again.
377	            LegVisibility::Closed
378	        } else {
379	            LegVisibility::Unconfirmed
380	        }
381	    }
382	
383	    fn reconcile_open_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
384	        for strategy_id in self.pending.strategy_ids() {
385	            if self.positions.has(&strategy_id) {
386	                continue;
387	            }
388	            let north_leg = match self.pending.leg(&strategy_id, Venue::North) {
389	                Some(order) => order.clone(),
390	                None => continue,
391	            };
392	            let south_leg = match self.pending.leg(&strategy_id, Venue::South) {
393	                Some(order) => order.clone(),
394	                None => continue,
395	            };
396	
397	            let north_complete = self.leg_is_complete(&north_leg);
398	            let south_complete = self.leg_is_complete(&south_leg);
399	
400	            if north_complete && south_complete {
401	                self.open_position(&north_leg, &south_leg, now);
402	            } else if north_complete {
403	                self.close_gap(&south_leg, now)?;
404	            } else if south_complete {
405	                self.close_gap(&north_leg, now)?;
406	            }
407	        }
408	        Ok(())
409	    }
410	
411	    /// Close the exposure left by a leg that did not complete.
412	    ///
413	    /// Correcting a resting leg is a two-step move. The cycle that first finds
414	    /// the exposure moves the resting order into closure, cancelling it at the
415	    /// venue; only a later cycle, once that cancellation boundary has passed,
416	    /// issues the corrective order for whatever residual then remains. The
417	    /// ordering exists so that a fill already in flight has a cycle in which to
418	    /// land and be attributed before another economic order is sent.
419	    ///
420	    /// An order the venue never confirmed is never neutralized against: it may
421	    /// still be resting and may still fill, and neutralizing quantity the desk
422	    /// already owns doubles the exposure the correction was meant to close.
423	    fn close_gap(&mut self, leg: &PendingOrder, now: Timestamp) -> Result<(), PortError> {
424	        match self.visibility(leg) {
425	            LegVisibility::Resting => {
426	                if !self.closing.contains(&leg.order_id) {
427	                    // First cycle for this leg: move it into closure and stop
428	                    // there. The corrective order waits for the next cycle.
429	                    if self.gateway.cancel(&leg.order_id).is_err() {
430	                        return Ok(());
431	                    }
432	                    self.closing.insert(leg.order_id.clone());
433	                    self.replaced.insert(leg.order_id.clone());
434	                    self.resting_at_venue.remove(&leg.order_id);
435	                    return Ok(());
436	                }
437	                // Already moved into closure on an earlier cycle. The venue may
438	                // still be listing it, but this engine has cancelled it, so it
439	                // is no longer working and the residual below is real.
440	            }
441	            // Absent from the live list and not withdrawn by this engine: its
442	            // state is unknown, so nothing is concluded and nothing is sent.
443	            LegVisibility::Unconfirmed => return Ok(()),
444	            // Withdrawn by this engine or closed by the venue: nothing is
445	            // resting, so the residual can be covered.
446	            LegVisibility::Withdrawn | LegVisibility::Closed => {}
447	        }
448	
449	        let residual = self.residual_for(leg);
450	        let plan = match plan_neutralization(
451	            self.gateway,
452	            leg,
453	            residual,
454	            self.config.neutralization_slippage,
455	        )? {
456	            Some(plan) => plan,
457	            None => return Ok(()),
458	        };
459	
460	        // The record is written before the order is sent, so a submission whose
461	        // outcome is unknown is still something this engine knows it committed
462	        // to. The submission is tracked under its `client_ref` for the same
463	        // reason: it is the key a later observation carries back.
464	        let client_ref = self.next_client_ref();
465	        let record = NeutralizationRecord {
466	            strategy_id: plan.strategy_id,
467	            venue: plan.venue,
468	            instrument_id: plan.instrument_id,
469	            client_ref: client_ref.clone(),
470	            order_id: None,
471	            limit_price: plan.limit_price,
472	            size: plan.size,
473	            expiry_time: leg.expiry_time,
474	        };
475	        let size = record.size;
476	        let request = Self::neutralization_request(&record);
477	        self.ledger
478	            .track(client_ref.clone(), None, None, size, now);
479	        self.neutralizations.push(record);
480	
481	        match self.gateway.submit(&request) {
482	            Ok(accepted) => {
483	                self.ledger.track(
484	                    client_ref.clone(),
485	                    Some(accepted.order_id.clone()),
486	                    Some(accepted.venue_seq),
487	                    size,
488	                    now,
489	                );
490	                if let Some(slot) = self
491	                    .neutralizations
492	                    .iter_mut()
493	                    .find(|slot| slot.client_ref == client_ref)
494	                {
495	                    slot.order_id = Some(accepted.order_id);
496	                }
497	            }
498	            // The venue may have created the order, so it stays committed: the
499	            // record keeps its `client_ref`, contributes its full size as
500	            // working, and is retried under that same reference.
501	            Err(SubmitError::AcknowledgementLost) => {}
502	            // Nothing was created, so nothing is working. Drop the record and
503	            // let a later cycle recompute the residual and try again: a mandate
504	            // is not latched shut by one failed attempt.
505	            Err(SubmitError::Rejected(_)) | Err(SubmitError::Transport(_)) => {
506	                self.neutralizations
507	                    .retain(|slot| slot.client_ref != client_ref);
508	            }
509	        }
510	
511	        Ok(())
512	    }

514	    fn open_position(&mut self, north_leg: &PendingOrder, south_leg: &PendingOrder, now: Timestamp) {
515	        let position = Position::new(
516	            north_leg.strategy_id.clone(),
517	            north_leg.size,
518	            south_leg.size,
519	            north_leg.limit_price,
520	            south_leg.limit_price,
521	            north_leg.expiry_time,
522	            now,
523	        );
524	        let strategy_id = position.strategy_id.clone();
525	        self.positions.insert(position);
526	        self.pending.remove(&north_leg.order_id);
527	        self.pending.remove(&south_leg.order_id);
528	        self.neutralizations
529	            .retain(|record| record.strategy_id != strategy_id);
530	    }
531	
532	    fn retire_stale_orders(&mut self, now: Timestamp) {
533	        for order_id in self.pending.expiring(now) {
534	            let _ = self.gateway.cancel(&order_id);
535	            self.pending.remove(&order_id);
536	            self.closing.remove(&order_id);
537	            self.replaced.remove(&order_id);
538	        }
539	    }
540	
541	    fn persist(&self) -> Result<(), PortError> {
542	        let snapshot = Snapshot {
543	            positions: self.positions.all(),
544	            pending_orders: self.pending.all(),
545	            attempted_strategies: self.attempted.iter().cloned().collect(),
546	            client_seq: self.client_seq,
547	            fill_cursor: self.ledger.cursor(),
548	            submissions: self.ledger.submissions(),
549	            absorbed_blocks: self.ledger.absorbed_blocks(),
550	            closing_orders: self.closing.iter().cloned().collect(),
551	            replaced_orders: self.replaced.iter().cloned().collect(),
552	            neutralizations: self.neutralizations.clone(),
553	        };
554	        self.store.save(&snapshot)
555	    }
556	
557	    fn next_client_ref(&mut self) -> ClientRef {
558	        self.client_seq += 1;
559	        ClientRef::new(format!(
560	            "{}-{}",
561	            self.config.client_ref_prefix, self.client_seq
562	        ))
563	    }
