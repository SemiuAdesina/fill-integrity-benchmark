318	    /// corrective order this engine sent for the same mandate and venue.
319	    ///
320	    /// A neutralization fill resolves the exposure, so it counts here once and
321	    /// is never also counted a second time as a fill of the resting order.
322	    fn acquired_for(&self, leg: &PendingOrder) -> Qty {
323	        let mut total = self.ledger.attributed_for(&leg.client_ref);
324	        for record in self.records_for(leg) {
325	            total += self.ledger.attributed_for(&record.client_ref);
326	        }

330	    /// Quantity still working in corrective orders already committed to for one
331	    /// leg.
332	    ///
333	    /// This is what stops consecutive cycles from covering the same exposure
334	    /// twice. A neutralization counts as working until it resolves:
335	    ///
336	    /// - one the venue has acknowledged contributes its own outstanding quantity;
337	    /// - one whose acknowledgement was lost contributes its whole size, because
338	    ///   the venue may have created it and this engine cannot yet see it;
339	    /// - one that has been finalized contributes nothing, whatever it managed to
340	    ///   fill, because the venue closed it and the gap it leaves behind is real.
341	    fn working_for(&self, leg: &PendingOrder) -> Qty {
342	        let mut working = 0;
343	        for record in self.records_for(leg) {
344	            match record.order_id {
345	                None => working += record.size,
346	                Some(_) if self.ledger.is_finalized(&record.client_ref) => {}
347	                Some(_) => {
348	                    let outstanding = record.size - self.ledger.attributed_for(&record.client_ref);
349	                    if outstanding > 0 {
350	                        working += outstanding;
351	                    }
352	                }
353	            }
354	        }
355	        working
356	    }

359	    /// attributable to it, less what is already working for it.

361	        leg.size - self.acquired_for(leg) - self.working_for(leg)
362	    }
