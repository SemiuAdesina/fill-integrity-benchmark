12	#[derive(Debug, Clone, PartialEq, Eq)]
13	pub enum PortError {
14	    Unavailable(String),
15	    Transport(String),
16	}
17	
18	impl fmt::Display for PortError {
19	    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
20	        match self {
21	            PortError::Unavailable(detail) => write!(f, "unavailable: {detail}"),
22	            PortError::Transport(detail) => write!(f, "transport: {detail}"),
23	        }
24	    }
25	}
26	
27	impl Error for PortError {}
28	
29	#[derive(Debug, Clone, PartialEq, Eq)]
30	pub enum SubmitError {
31	    Rejected(String),
32	    AcknowledgementLost,
33	    Transport(String),
34	}
35	
36	impl fmt::Display for SubmitError {
37	    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
38	        match self {
39	            SubmitError::Rejected(detail) => write!(f, "rejected: {detail}"),
40	            SubmitError::AcknowledgementLost => write!(f, "acknowledgement lost"),
41	            SubmitError::Transport(detail) => write!(f, "transport: {detail}"),
42	        }
43	    }
44	}
45	
46	impl Error for SubmitError {}

104	/// One block match: several of the engine's own orders on one instrument and
105	/// venue matched together and reported as a single netted quantity.
106	///
107	/// The venue names every order the block was matched against in `covers` and
108	/// reports only the block total in `block_qty`. It does not say how the total
109	/// divides. The division rule is in `/app/spec/venue-semantics.md`.
110	#[derive(Debug, Clone, PartialEq)]
111	pub struct BlockFill {
112	    pub seq: u64,
113	    /// Idempotency key for the block. The same `block_ref` is the same match,
114	    /// however many times it is delivered.
115	    pub block_ref: BlockRef,
116	    pub instrument_id: InstrumentId,
117	    pub venue: Venue,
118	    pub covers: Vec<OrderId>,
119	    /// Total base units matched by this block, across every covered order.
120	    pub block_qty: Qty,
121	    pub price: Price,
122	    pub at: Timestamp,
123	}

128	    pub blocks: Vec<BlockFill>,

130	}
