2	//! One engine cycle: read the venue, decide what completed, act, persist.
3	//!
4	//! `run_cycle` is the only entry point the graded suite calls. The order of the
5	//! steps below is the order they run in, and everything else in the crate is a
6	//! type or a helper reached from here.
7

26	/// What the venue currently shows about a leg the engine tracks.
27	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28	enum LegVisibility {
29	    /// Present in `open_orders`: still working, so nothing may be concluded
30	    /// from it yet.
31	    Resting,
32	    /// Absent from `open_orders`, and this engine cancelled it.
33	    Withdrawn,
34	    /// Absent from `open_orders`, and the venue has closed it: a block named
35	    /// it. It will never acquire another base unit, so what it is short is
36	    /// known.
37	    Closed,
38	    /// Absent from `open_orders` for a reason this engine cannot see. It may
39	    /// still be resting, and a fill may still be in flight, so nothing may be
40	    /// concluded from its absence.
41	    Unconfirmed,
42	}
43	
44	pub struct Engine<'a> {
45	    config: EngineConfig,
46	    strategies: &'a dyn StrategySource,
47	    inventory: &'a dyn InventorySource,
48	    gateway: &'a dyn OrderGateway,
49	    fills: &'a dyn FillEventSource,
50	    store: &'a dyn StateStore,
51	    clock: &'a dyn Clock,
52	    pending: PendingOrderManager,
53	    positions: PositionManager,
54	    attempted: BTreeSet<StrategyId>,
55	    /// Per-order attribution state, and the feed cursor that produced it.
56	    ledger: AttributionLedger,
57	    /// Corrective orders this engine has committed to, resolved or not.
58	    neutralizations: Vec<NeutralizationRecord>,
59	    /// Legs entered but not yet confirmed at the venue.
60	    leg_attempts: Vec<LegAttempt>,
61	    /// Legs marked for closure on an earlier cycle.
62	    closing: BTreeSet<OrderId>,
63	    /// Legs this engine has cancelled at the venue.
64	    replaced: BTreeSet<OrderId>,
65	    /// Order ids the last `open_orders` call reported as live. Not durable: it is
66	    /// re-read every cycle and says nothing after a restart.
67	    resting_at_venue: BTreeSet<OrderId>,
68	    loaded: bool,
69	    client_seq: u64,
70	    observed_fill_events: usize,
71	}

119	    pub fn run_cycle(&mut self) -> Result<(), PortError> {
120	        self.ensure_loaded()?;
121	        let now = self.clock.now();
122	        self.sync_observations();
123	        self.reconcile_neutralizations(now);
124	        self.enter_new_strategies(now)?;
125	        self.retry_unconfirmed_legs(now);
126	        self.reconcile_open_strategies(now)?;
127	        self.retire_stale_orders(now);
128	        self.persist()?;
129	        Ok(())
130	    }
131	
132	    /// Restore the durable state on the first cycle after a start or restart.
133	    fn ensure_loaded(&mut self) -> Result<(), PortError> {
134	        if self.loaded {
135	            return Ok(());
136	        }
137	        if let Some(snapshot) = self.store.load()? {
138	            self.positions = PositionManager::from_positions(snapshot.positions);
139	            self.pending = PendingOrderManager::from_orders(snapshot.pending_orders);
140	            self.attempted = snapshot.attempted_strategies.into_iter().collect();
141	            self.client_seq = snapshot.client_seq;
142	            self.ledger = AttributionLedger::restore(
143	                snapshot.submissions,
144	                snapshot.absorbed_blocks,
145	                snapshot.fill_cursor,
146	            );
147	            self.closing = snapshot.closing_orders.into_iter().collect();
148	            self.replaced = snapshot.replaced_orders.into_iter().collect();
149	            self.neutralizations = snapshot.neutralizations;
150	            self.leg_attempts = snapshot.leg_attempts;
151	        }
152	        self.loaded = true;
153	        Ok(())
154	    }
155	
156	    /// Read both venue sources into the ledger before anything is decided.
157	    ///
158	    /// A failed `open_orders` says nothing about the venue, so no order can be
159	    /// assumed live from it.
160	    fn sync_observations(&mut self) {
161	        match self.gateway.open_orders() {
162	            Ok(open) => {
163	                self.ledger.absorb_open_orders(&open);
164	                self.resting_at_venue = open.into_iter().map(|order| order.order_id).collect();
165	            }
166	            Err(_) => {
167	                self.resting_at_venue.clear();
168	            }
169	        }
170	        if let Ok(batch) = self.fills.poll(&self.ledger.cursor()) {
171	            self.observed_fill_events += self.ledger.absorb_batch(batch);
172	        }
173	    }
174	
175	    /// Match up corrective orders whose fate was unknown when they were sent.
176	    ///
177	    /// A record without an `order_id` is one the venue never acknowledged. The
178	    /// client reference is the venue's idempotency key, so resubmitting under it
179	    /// returns the original order rather than creating a second one.
180	    fn reconcile_neutralizations(&mut self, now: Timestamp) {
181	        // Identity first. A record with no `order_id` is one the venue never
182	        // acknowledged; the venue echoes the `client_ref` back on the live order
183	        // list and on fill events, so an observation may have named it since.
184	        let mut resolved: Vec<(ClientRef, OrderId)> = Vec::new();
185	        for record in &self.neutralizations {
186	            if record.order_id.is_some() {
187	                continue;
188	            }
189	            if let Some(order_id) = self.ledger.order_for(&record.client_ref) {
190	                resolved.push((record.client_ref.clone(), order_id));
191	            }
192	        }
193	        for (client_ref, order_id) in resolved {
194	            if let Some(slot) = self
195	                .neutralizations
196	                .iter_mut()
197	                .find(|slot| slot.client_ref == client_ref)
198	            {
199	                slot.order_id = Some(order_id);
200	            }
201	        }
202	
203	        // Whatever is still unacknowledged is retried under the same reference.
204	        // The venue deduplicates on it, so the retry resolves onto the original
205	        // order if the venue created one, and creates it if it did not. A
206	        // freshly minted reference would place the same exposure twice.
207	        let unconfirmed: Vec<NeutralizationRecord> = self
208	            .neutralizations
209	            .iter()
210	            .filter(|record| record.order_id.is_none())
211	            .cloned()
212	            .collect();
213	        for record in unconfirmed {
214	            let request = Self::neutralization_request(&record);
215	            if let Ok(accepted) = self.gateway.submit(&request) {
216	                self.ledger.track(
217	                    record.client_ref.clone(),
218	                    Some(accepted.order_id.clone()),
219	                    Some(accepted.venue_seq),
220	                    record.size,
221	                    now,
222	                );
223	                if let Some(slot) = self
224	                    .neutralizations
225	                    .iter_mut()
226	                    .find(|slot| slot.client_ref == record.client_ref)
227	                {
228	                    slot.order_id = Some(accepted.order_id);
229	                }
230	            }
231	        }
232	    }
233	
234	    fn neutralization_request(record: &NeutralizationRecord) -> OrderRequest {
235	        OrderRequest {
236	            strategy_id: record.strategy_id.clone(),
237	            instrument_id: record.instrument_id.clone(),
238	            venue: record.venue,
239	            kind: OrderKind::ImmediateOrCancel,
240	            limit_price: record.limit_price,
241	            size: record.size,
242	            client_ref: record.client_ref.clone(),
243	            expiry_time: record.expiry_time,
244	        }
245	    }

247	    fn enter_new_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
248	        let available_cash = self.inventory.settlement_balance()?;
249	        let candidates = self.strategies.candidates()?;
250	        for candidate in candidates {
251	            if self.positions.has(&candidate.strategy_id) {
252	                continue;
253	            }
254	            if self.attempted.contains(&candidate.strategy_id) {
255	                continue;
256	            }
257	            if candidate.pair_cost() > self.config.max_pair_cost {
258	                continue;
259	            }
260	            if now >= candidate.expiry_time {
261	                continue;
262	            }
263	            let north_size = leg_size(candidate.north_depth, self.config.max_leg_size);
264	            let south_size = leg_size(candidate.south_depth, self.config.max_leg_size);
265	            if north_size < self.config.min_leg_size || south_size < self.config.min_leg_size {
266	                continue;
267	            }
268	            let notional = (candidate.north_price * north_size as f64)
269	                + (candidate.south_price * south_size as f64);
270	            if notional > available_cash {
271	                continue;
272	            }
273	            self.attempted.insert(candidate.strategy_id.clone());
274	            self.place_leg(&candidate, Venue::North, north_size, now);
275	            self.place_leg(&candidate, Venue::South, south_size, now);
276	        }
277	        Ok(())
278	    }
279	
280	    fn place_leg(&mut self, candidate: &StrategyCandidate, venue: Venue, size: Qty, now: Timestamp) {
281	        let client_ref = self.next_client_ref();
282	        let request = OrderRequest {
283	            strategy_id: candidate.strategy_id.clone(),
284	            instrument_id: candidate.instrument_for(venue).clone(),
285	            venue,
286	            kind: OrderKind::RestingLimit,
287	            limit_price: candidate.price_for(venue),
288	            size,
289	            client_ref: client_ref.clone(),
290	            expiry_time: candidate.expiry_time,
291	        };
292	        match self.gateway.submit(&request) {
293	            Ok(accepted) => {
294	                self.resting_at_venue.insert(accepted.order_id.clone());
295	                self.ledger.track(
296	                    client_ref.clone(),
297	                    Some(accepted.order_id.clone()),
298	                    Some(accepted.venue_seq),
299	                    size,
300	                    now,
301	                );
302	                self.pending.insert(PendingOrder::new(
303	                    candidate.strategy_id.clone(),
304	                    candidate.instrument_for(venue).clone(),
305	                    venue,
306	                    accepted.order_id,
307	                    client_ref,
308	                    accepted.venue_seq,
309	                    request.limit_price,
310	                    size,
311	                    candidate.expiry_time,
312	                    now,
313	                ));
314	            }
315	            // Neither a rejection nor a lost acknowledgement is a settled leg.
316	            // The identity the retry depends on is the reference the engine
317	            // chose, so it is tracked now and reused: a fresh reference would
318	            // risk a second economic order for the same leg.
319	            Err(_) => {
320	                self.ledger
321	                    .track(client_ref.clone(), None, None, size, now);
322	                self.leg_attempts.push(LegAttempt {
323	                    strategy_id: candidate.strategy_id.clone(),
324	                    instrument_id: candidate.instrument_for(venue).clone(),
325	                    venue,
326	                    limit_price: request.limit_price,
327	                    size,
328	                    expiry_time: candidate.expiry_time,
329	                    client_ref,
330	                    created_at: now,
331	                });
332	            }
333	        }
334	    }
335	
336	    /// Re-enter legs the venue never confirmed, under the reference they were
337	    /// first sent under.
338	    ///
339	    /// A leg is recovered either by identity — an observation has since carried
340	    /// the `client_ref` back, which is what makes an unacknowledged order
341	    /// visible again — or by resubmitting under that same reference, which the
342	    /// venue deduplicates onto the original order. Nothing here mints a new
343	    /// reference, and a leg that is already tracked is left alone.
344	    fn retry_unconfirmed_legs(&mut self, now: Timestamp) {
345	        if self.leg_attempts.is_empty() {
346	            return;
347	        }
348	        let attempts = std::mem::take(&mut self.leg_attempts);
349	        let mut still_unconfirmed: Vec<LegAttempt> = Vec::new();
350	        for attempt in attempts {
351	            // Expired: there is nothing left to work.
352	            if now >= attempt.expiry_time {
353	                continue;
354	            }
355	            if self.pending.leg(&attempt.strategy_id, attempt.venue).is_some() {
356	                continue;
357	            }
358	            if let Some(order_id) = self.ledger.order_for(&attempt.client_ref) {
359	                let venue_seq = self.ledger.venue_seq_for(&attempt.client_ref).unwrap_or(0);
360	                self.pending.insert(PendingOrder::new(
361	                    attempt.strategy_id.clone(),
362	                    attempt.instrument_id.clone(),
363	                    attempt.venue,
364	                    order_id,
365	                    attempt.client_ref.clone(),
366	                    venue_seq,
367	                    attempt.limit_price,
368	                    attempt.size,
369	                    attempt.expiry_time,
370	                    attempt.created_at,
371	                ));
372	                continue;
373	            }
374	            let request = OrderRequest {
375	                strategy_id: attempt.strategy_id.clone(),
376	                instrument_id: attempt.instrument_id.clone(),
377	                venue: attempt.venue,
378	                kind: OrderKind::RestingLimit,
379	                limit_price: attempt.limit_price,
380	                size: attempt.size,
381	                client_ref: attempt.client_ref.clone(),
382	                expiry_time: attempt.expiry_time,
383	            };
384	            match self.gateway.submit(&request) {
385	                Ok(accepted) => {
386	                    self.resting_at_venue.insert(accepted.order_id.clone());
387	                    self.ledger.track(
388	                        attempt.client_ref.clone(),
389	                        Some(accepted.order_id.clone()),
390	                        Some(accepted.venue_seq),
391	                        attempt.size,
392	                        attempt.created_at,
393	                    );
394	                    self.pending.insert(PendingOrder::new(
395	                        attempt.strategy_id.clone(),
396	                        attempt.instrument_id.clone(),
397	                        attempt.venue,
398	                        accepted.order_id,
399	                        attempt.client_ref.clone(),
400	                        accepted.venue_seq,
401	                        attempt.limit_price,
402	                        attempt.size,
403	                        attempt.expiry_time,
404	                        attempt.created_at,
405	                    ));
406	                }
407	                Err(_) => still_unconfirmed.push(attempt),
408	            }
409	        }
410	        self.leg_attempts = still_unconfirmed;
411	    }

413	    /// Quantity attributable to this leg: the resting order itself, plus every
414	    /// corrective order this engine sent for the same mandate and venue.
415	    fn acquired_for(&self, leg: &PendingOrder) -> Qty {
416	        let mut total = self.ledger.attributed_for(&leg.client_ref);
417	        for record in self.records_for(leg) {
418	            total += self.ledger.attributed_for(&record.client_ref);
419	        }
420	        total
421	    }
422	
423	    fn records_for(&self, leg: &PendingOrder) -> Vec<&NeutralizationRecord> {
424	        self.neutralizations
425	            .iter()
426	            .filter(|record| record.strategy_id == leg.strategy_id && record.venue == leg.venue)
427	            .collect()
428	    }
429	
430	    /// Exact integer comparison. There is no tolerance: a leg short by one base
431	    /// unit is short.
432	    fn leg_is_complete(&self, leg: &PendingOrder) -> bool {
433	        self.acquired_for(leg) >= leg.size
434	    }
435	
436	    /// Quantity already working on corrective orders this engine has committed
437	    /// to for this leg.
438	    ///
439	    /// This is the term that stops consecutive cycles sending the same exposure
440	    /// twice:
441	    ///
442	    /// - one the venue acknowledged is working for its own outstanding quantity;
443	    /// - one whose acknowledgement was lost is working for its full size, since
444	    ///   the venue may have created it and this engine cannot see it;
445	    /// - one the venue has finalized is working for nothing at all, whatever it
446	    ///   managed to fill, because the gap it leaves behind is real.
447	    fn working_for(&self, leg: &PendingOrder) -> Qty {
448	        self.records_for(leg)
449	            .iter()
450	            .map(|record| self.working_in(record))
451	            .sum()
452	    }
453	
454	    fn working_in(&self, record: &NeutralizationRecord) -> Qty {
455	        if record.order_id.is_none() {
456	            return record.size;
457	        }
458	        if !self.ledger.knows(&record.client_ref) {
459	            return record.size;
460	        }
461	        self.ledger.outstanding_for(&record.client_ref)
462	    }
463	
464	    /// The exposure on this leg that is not yet resolved: what the leg needs,
465	    /// less what is already attributed to it, less what is still working in
466	    /// corrective orders already committed to.
467	    fn residual_for(&self, leg: &PendingOrder) -> Qty {
468	        leg.size - self.acquired_for(leg) - self.working_for(leg)
469	    }
470	
471	    fn visibility(&self, leg: &PendingOrder) -> LegVisibility {
472	        if self.resting_at_venue.contains(&leg.order_id) {
473	            LegVisibility::Resting
474	        } else if self.ledger.is_finalized(&leg.client_ref) {
475	            // A block named this order, so the venue has closed it and
476	            // cancelled the rest: what it is short is known, not in flight.
477	            LegVisibility::Closed
478	        } else if self.replaced.contains(&leg.order_id) {
479	            LegVisibility::Withdrawn
480	        } else {
481	            LegVisibility::Unconfirmed
482	        }
483	    }

485	    fn reconcile_open_strategies(&mut self, now: Timestamp) -> Result<(), PortError> {
486	        for strategy_id in self.pending.strategy_ids() {
487	            if self.positions.has(&strategy_id) {
488	                continue;
489	            }
490	            let north_leg = match self.pending.leg(&strategy_id, Venue::North) {
491	                Some(order) => order.clone(),
492	                None => continue,
493	            };
494	            let south_leg = match self.pending.leg(&strategy_id, Venue::South) {
495	                Some(order) => order.clone(),
496	                None => continue,
497	            };
498	
499	            let north_complete = self.leg_is_complete(&north_leg);
500	            let south_complete = self.leg_is_complete(&south_leg);
501	
502	            if north_complete && south_complete {
503	                self.open_position(&north_leg, &south_leg, now);
504	            } else if north_complete {
505	                self.close_gap(&south_leg, now)?;
506	            } else if south_complete {
507	                self.close_gap(&north_leg, now)?;
508	            }
509	        }
510	        Ok(())
511	    }
512	
513	    /// Close the exposure left by a leg that did not complete.
514	    ///
515	    /// Closing a resting leg is a two-step move: the first cycle marks it, the
516	    /// next one cancels it and covers what is missing. That gives an in-flight
517	    /// fill a cycle to arrive before another economic order is sent.
518	    fn close_gap(&mut self, leg: &PendingOrder, now: Timestamp) -> Result<(), PortError> {
519	        match self.visibility(leg) {
520	            LegVisibility::Resting => {
521	                if self.closing.insert(leg.order_id.clone()) {
522	                    return Ok(());
523	                }
524	                if self.gateway.cancel(&leg.order_id).is_err() {
525	                    return Ok(());
526	                }
527	                self.resting_at_venue.remove(&leg.order_id);
528	                self.replaced.insert(leg.order_id.clone());
529	            }
530	            // The venue has closed it, or this engine withdrew it: there is
531	            // nothing to cancel first, and what it is short is known.
532	            LegVisibility::Closed | LegVisibility::Withdrawn => {}
533	            // Its absence from the live list proves nothing: it may still be
534	            // resting and a fill may still be in flight. Neutralizing now would
535	            // acquire quantity the desk already owns, doubling the exposure the
536	            // correction was meant to close.
537	            LegVisibility::Unconfirmed => return Ok(()),
538	        }
539	
540	        let gap = self.residual_for(leg);
541	        let plan = match plan_neutralization(self.gateway, leg, gap, self.config.neutralization_slippage)? {
542	            Some(plan) => plan,
543	            None => return Ok(()),
544	        };
545	
546	        // The record is written before the order is sent, so a submission whose
547	        // outcome is unknown is still something this engine knows it committed to.
548	        let client_ref = self.next_client_ref();
549	        let record = NeutralizationRecord {
550	            strategy_id: plan.strategy_id,
551	            venue: plan.venue,
552	            instrument_id: plan.instrument_id,
553	            client_ref: client_ref.clone(),
554	            order_id: None,
555	            limit_price: plan.limit_price,
556	            size: plan.size,
557	            expiry_time: leg.expiry_time,
558	        };
559	        let size = record.size;
560	        let request = Self::neutralization_request(&record);
561	        self.neutralizations.push(record);
562	        self.ledger
563	            .track(client_ref.clone(), None, None, size, now);
564	
565	        match self.gateway.submit(&request) {
566	            Ok(accepted) => {
567	                self.ledger.learn_order(
568	                    &client_ref,
569	                    accepted.order_id.clone(),
570	                    Some(accepted.venue_seq),
571	                );
572	                if let Some(slot) = self
573	                    .neutralizations
574	                    .iter_mut()
575	                    .find(|slot| slot.client_ref == client_ref)
576	                {
577	                    slot.order_id = Some(accepted.order_id);
578	                }
579	            }
580	            // The venue may have created it under an identifier this engine
581	            // cannot see. The record stays, so it counts as working and the
582	            // retry reuses the reference it was sent under.
583	            Err(SubmitError::AcknowledgementLost) => {}
584	            // Nothing was created, so there is nothing committed to. Drop the
585	            // record and let a later cycle recompute the residual and try again:
586	            // one failed attempt must not latch the mandate shut.
587	            Err(_) => {
588	                self.neutralizations
589	                    .retain(|slot| slot.client_ref != client_ref);
590	            }
591	        }
592	
593	        Ok(())
594	    }

596	    fn open_position(&mut self, north_leg: &PendingOrder, south_leg: &PendingOrder, now: Timestamp) {
597	        let position = Position::new(
598	            north_leg.strategy_id.clone(),
599	            north_leg.size,
600	            south_leg.size,
601	            north_leg.limit_price,
602	            south_leg.limit_price,
603	            north_leg.expiry_time,
604	            now,
605	        );
606	        let strategy_id = position.strategy_id.clone();
607	        self.positions.insert(position);
608	        self.closing.remove(&north_leg.order_id);
609	        self.closing.remove(&south_leg.order_id);
610	        self.replaced.remove(&north_leg.order_id);
611	        self.replaced.remove(&south_leg.order_id);
612	        self.pending.remove(&north_leg.order_id);
613	        self.pending.remove(&south_leg.order_id);
614	        self.neutralizations
615	            .retain(|record| record.strategy_id != strategy_id);
616	        self.leg_attempts
617	            .retain(|attempt| attempt.strategy_id != strategy_id);
618	    }
619	
620	    fn retire_stale_orders(&mut self, now: Timestamp) {
621	        for order_id in self.pending.expiring(now) {
622	            let _ = self.gateway.cancel(&order_id);
623	            self.pending.remove(&order_id);
624	            self.closing.remove(&order_id);
625	            self.replaced.remove(&order_id);
626	        }
627	        self.leg_attempts
628	            .retain(|attempt| now < attempt.expiry_time);
629	    }

631	    fn persist(&self) -> Result<(), PortError> {
632	        let snapshot = Snapshot {
633	            positions: self.positions.all(),
634	            pending_orders: self.pending.all(),
635	            attempted_strategies: self.attempted.iter().cloned().collect(),
636	            client_seq: self.client_seq,
637	            fill_cursor: self.ledger.cursor(),
638	            submissions: self.ledger.submissions(),
639	            absorbed_blocks: self.ledger.absorbed_blocks(),
640	            closing_orders: self.closing.iter().cloned().collect(),
641	            replaced_orders: self.replaced.iter().cloned().collect(),
642	            neutralizations: self.neutralizations.clone(),
643	            leg_attempts: self.leg_attempts.clone(),
644	        };
645	        self.store.save(&snapshot)
646	    }
