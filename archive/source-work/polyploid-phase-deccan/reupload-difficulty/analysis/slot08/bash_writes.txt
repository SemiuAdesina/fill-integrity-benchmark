import json, random, sys

def gen(markers, offspring, free, slope, planted, seed):
    rnd = random.Random(seed)
    def cols(d):
        return [b for b in range(16) if bin(b).count('1')==d]
    def peaked(truth):
        return [-slope*abs(d-truth) - 0.01*d for d in range(5)]
    dosage_a=[0]*markers; dosage_b=[0]*markers
    for i in rnd.sample(range(markers), free): dosage_a[i]=rnd.randint(1,3)
    for i in rnd.sample(range(markers), free): dosage_b[i]=rnd.randint(1,3)
    cola=[rnd.choice(cols(d)) for d in dosage_a]
    colb=[rnd.choice(cols(d)) for d in dosage_b]
    states=[(a,b) for a in range(4) for b in range(4) if a<=b]
    gl=[]
    for c in range(offspring):
        rows=[]
        for m in range(markers):
            sa=rnd.choice([s for s in states if s[0]!=s[1]])
            sb=rnd.choice(states)
            g=((cola[m]>>sa[0])&1)+((cola[m]>>sa[1])&1)+((colb[m]>>sb[0])&1)+((colb[m]>>sb[1])&1)
            row = peaked(g) if planted else [-3.0*rnd.random() for _ in range(5)]
            rows.append([round(x,6) for x in row])
        gl.append(rows)
    rec_a=[round(0.05+0.2*rnd.random(),6) for _ in range(markers-1)]
    rec_b=[round(0.05+0.2*rnd.random(),6) for _ in range(markers-1)]
    al_a=[rnd.random()<0.5 for _ in range(markers)]
    al_b=[rnd.random()<0.5 for _ in range(markers)]
    ra=[round(0.1+0.3*rnd.random(),6) if a else 0.0 for a in al_a]
    rb=[round(0.1+0.3*rnd.random(),6) if a else 0.0 for a in al_b]
    return {"format_version":1,"markers":markers,"offspring":offspring,
            "parent_dosage":{"A":dosage_a,"B":dosage_b},"child_gl":gl,
            "recombination":{"A":rec_a,"B":rec_b},
            "double_reduction":{"A":{"allowed":al_a,"rate":ra},"B":{"allowed":al_b,"rate":rb}}}

markers=int(sys.argv[1]); offspring=int(sys.argv[2]); free=int(sys.argv[3])
