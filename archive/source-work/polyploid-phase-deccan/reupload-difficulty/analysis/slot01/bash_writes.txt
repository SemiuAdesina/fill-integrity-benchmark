pub mod local;
pub mod tables;
pub mod viterbi;

use crate::cross_input::Cross;
use crate::cross_model::{Haplotypes, Parent, ProgenyPath, Reconstruction, State};
use crate::solver::local::{Local, Rng};
use crate::solver::tables::Tables;
use crate::solver::viterbi::{Chain, SuffixBounds};

pub fn solve(cross: &Cross) -> Reconstruction {
    let tables = Tables::build(cross);
    let _bounds = SuffixBounds::build(cross, &tables);
    let mut rng = Rng::new(0x5eed);
    let mut local = Local::new(cross, &tables);
    let mut best = f64::NEG_INFINITY;
    let mut best_picks: Vec<(usize, usize)> = Vec::new();
    for _ in 0..8 {
        let mut picks: Vec<(usize, usize)> = (0..cross.markers)
            .map(|m| (rng.below(tables.cols[0][m].len()), rng.below(tables.cols[1][m].len())))
            .collect();
        let v = local.ascend(&mut picks);
        if v > best {
            best = v;
            best_picks = picks;
        }
    }
    let _ = best;
    let chain = Chain::new(cross, &tables);
    let mut hap_a = Haplotypes::zeros(cross.markers);
    let mut hap_b = Haplotypes::zeros(cross.markers);
    for m in 0..cross.markers {
        let mask_a = tables.cols[0][m][best_picks[m].0];
        let mask_b = tables.cols[1][m][best_picks[m].1];
        for h in 0..4 {
            hap_a.set(h, m, (mask_a >> h) & 1);
            hap_b.set(h, m, (mask_b >> h) & 1);
        }
    }
    let mut paths = Vec::with_capacity(cross.offspring);
    for c in 0..cross.offspring {
        let (_, joint) = chain.best_path(c, &best_picks);
        let mut from_a = Vec::with_capacity(cross.markers);
        let mut from_b = Vec::with_capacity(cross.markers);
        for (m, j) in joint.iter().enumerate() {
            let (sa, sb) = local::split(*j);
            let _ = m;
            from_a.push(state_of(sa));
            from_b.push(state_of(sb));
        }
        paths.push(ProgenyPath { from_a, from_b });
    }
    let _ = Parent::A;
    Reconstruction { haplotypes_a: hap_a, haplotypes_b: hap_b, paths }
}

fn state_of(index: usize) -> State {
    crate::cross_model::ALL_STATES[index]
}
