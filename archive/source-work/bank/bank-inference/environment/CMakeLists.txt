cmake_minimum_required(VERSION 3.16)
project(bank_inference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bank_model src/model.cpp src/recover.cpp)
target_include_directories(bank_model PUBLIC include)
target_compile_options(bank_model PRIVATE -Wall -Wextra -Wpedantic)
add_executable(bank_demo examples/demo.cpp)
target_link_libraries(bank_demo PRIVATE bank_model)
add_executable(bank_diagnostics tests/diagnostics.cpp)
target_link_libraries(bank_diagnostics PRIVATE bank_model)
enable_testing()
add_test(NAME diagnostics COMMAND bank_diagnostics)
