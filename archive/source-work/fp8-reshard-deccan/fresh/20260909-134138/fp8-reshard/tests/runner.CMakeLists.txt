# Candidate-linked runner build: the verifier's frozen sources, the probe
# driver, and the allow-listed candidate restoration. No judge target, no
# certified answer.
cmake_minimum_required(VERSION 3.16)
project(fp8_verifier_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CANDIDATE_SOURCES CONFIGURE_DEPENDS candidate/restore.cpp candidate/restore/*.cpp)
add_executable(probe src/json.cpp src/model.cpp src/probe.cpp ${CANDIDATE_SOURCES})
target_include_directories(probe PRIVATE include src candidate)
target_compile_options(probe PRIVATE -O2 -Wall -Wextra)
