#include "roundel/runtime.hpp"
#include <cstdio>
using namespace roundel;
static int fails = 0;
#define CK(c) do { if(!(c)) { ++fails; std::printf("  FAIL %d: %s\n", __LINE__, #c); } } while(0)

static Duty self_poster(Conductor* c, JobId self) {
    c->post(self, 111);  // queued while running: binds no frame
    const std::int64_t v = co_await park();
    co_return v;
}
static Duty self_poster_after(Conductor* c, JobId self) {
    c->post(self, 111);
    const std::int64_t a = co_await park();
    const std::int64_t b = co_await park();  // parks again in the same frame
    co_return a * 100 + b;
}
int main() {
    setvbuf(stdout, nullptr, _IONBF, 0);
    {
        Conductor c{4};
        const JobId a = c.launch(self_poster(&c, 1));
        c.post(a, 7);  // queued while parked
        c.drain();
        CK(c.settled(a));
        CK(*c.value_of(a) == 7);
        CK(c.journal().count_for(Mark::TicketRejected, a) == 1);
        CK(c.journal().count_for(Mark::TicketAccepted, a) == 1);
        CK(c.arena().live_slots() == 0);
        std::printf("unbound-then-parked: value=%lld rejected=%zu accepted=%zu\n",
                    (long long)*c.value_of(a), c.journal().count_for(Mark::TicketRejected, a),
                    c.journal().count_for(Mark::TicketAccepted, a));
    }
    {
        // same, but the job parks twice: the unbound ticket stays unbound
        Conductor c{4};
        const JobId a = c.launch(self_poster_after(&c, 1));
        c.post(a, 7);
        c.post(a, 8);
        c.drain();
        CK(c.settled(a));
        CK(*c.value_of(a) == 708);
        CK(c.journal().count_for(Mark::TicketRejected, a) == 1);
        CK(c.journal().count_for(Mark::TicketAccepted, a) == 2);
        std::printf("unbound+two parks: value=%lld\n", (long long)*c.value_of(a));
    }
    std::printf("%s (%d failures)\n", fails ? "FAILED" : "ok", fails);
    return fails ? 1 : 0;
}
EOF
