class Conductor {
  public:
    // `capacity` frames may be checked out at the same time.
    explicit Conductor(std::size_t capacity = 8);
    ~Conductor();

    Conductor(const Conductor&) = delete;
    Conductor& operator=(const Conductor&) = delete;

    // Adopt a freshly made Duty as a root job and run it to its first
    // suspension. Returns the job's identifier, which is never reused.
    JobId launch(Duty duty);

    // The conductor's logical clock. It starts at 0 and only ever moves
    // forward, to the ready time of the ticket a step delivers.
    std::uint64_t now() const noexcept { return tick_; }

    // Queue a completion for `job`, ready at now(), to be delivered by a later
    // step().
    void post(JobId job, std::int64_t value);
    // Queue a completion for `job` ready at now() + delay.
    void post_after(JobId job, std::int64_t value, std::uint64_t delay);
    // Queue a completion that raises an exception where `job` is parked, ready
    // at now(). A job that does not catch it ends up Faulted.
    void post_fault(JobId job);
    // The same, ready at now() + delay.
    void post_fault_after(JobId job, std::uint64_t delay);

    // Ask `job` to stop. Calling this more than once, or for a job that has
    // already settled, must be harmless.
    void cancel(JobId job);

    // Deliver at most one queued ticket: the one that comes first in the
    // delivery order (earliest ready time; then lowest job identifier; then
    // earliest queued). Returns false when the queue was empty. Nothing else
    // runs.
    bool step();
    // Deliver queued tickets until the queue is empty. Returns how many ran.
    std::size_t drain();

    std::size_t pending_tickets() const noexcept { return queue_.size(); }

    // Outcome of a settled job, or nullopt while it is still running.
    std::optional<Outcome> outcome_of(JobId job) const;
    // Value produced by a job that delivered one.

    bool settled(JobId job) const;

// -------------------------------------------------------------------------
// Conductor
// -------------------------------------------------------------------------

Conductor::Conductor(std::size_t capacity) : arena_(capacity, 512) {
    arena_.observe(&journal_);
    previous_arena_ = g_arena;
    g_arena = &arena_;
}

Conductor::~Conductor() {
    // Anything still open is stopped and its frames destroyed, lowest
    // identifier first, so the arena is whole when it goes away.
    std::vector<JobId> ids;
    ids.reserve(jobs_.size());
    for (const auto& kv : jobs_) {
        ids.push_back(kv.first);
    }
    std::sort(ids.begin(), ids.end());
    for (JobId job : ids) {
        const JobState* st = find(job);
        if (st != nullptr && !st->settled) {
            settle(job, Outcome::Cancelled, 0);
        }
    }
    flush_releases();
    for (JobId job : ids) {
        JobState* st = find(job);
        if (st == nullptr || !st->root) {
            continue;
        }
        std::coroutine_handle<> r = std::exchange(st->root, {});
        st->parked = {};
        st->parked_slot = SlotId{};
        const SlotId slot = FrameArena::slot_of(r.address());
        r.destroy();
        journal_.record_slot(Mark::FrameReleased, job, slot);
    }
    g_arena = previous_arena_;
}

FrameArena* Conductor::active_arena() noexcept { return g_arena; }

Conductor::JobState* Conductor::find(JobId job) {
    auto it = jobs_.find(job);
    return (it == jobs_.end()) ? nullptr : &it->second;
}

const Conductor::JobState* Conductor::find(JobId job) const {
    auto it = jobs_.find(job);
    return (it == jobs_.end()) ? nullptr : &it->second;
}

JobId Conductor::launch(Duty duty) {
    Duty::handle_type h = duty.release();
    if (!h) {
        return kNoJob;
    }
    const JobId job = next_job_id();
    h.promise().conductor = this;
    h.promise().job = job;

    JobState st;
    st.root = h;
    JobState* state = &jobs_.emplace(job, st).first->second;

    journal_.record_slot(Mark::JobStarted, job, FrameArena::slot_of(h.address()));

    resume_frame(state, h);
    return job;
}

void Conductor::note_parked(JobId job, std::coroutine_handle<> h) {
    JobState* st = find(job);
    if (st == nullptr || st->settled) {
        return;
    }
    st->parked = h;
    st->parked_slot = FrameArena::slot_of(h.address());
    journal_.record_slot(Mark::JobParked, job, st->parked_slot);
}

void Conductor::settle(JobId job, Outcome outcome, std::int64_t value) {
    JobState* st = find(job);
    if (st == nullptr || st->settled) {
        // One outcome per job: whatever else arrives afterwards changes
        // nothing.
        return;
    }
    st->settled = true;
    st->outcome = outcome;
    st->value = value;
    st->parked = {};
    st->parked_slot = SlotId{};
    journal_.record(Mark::JobSettled, job, static_cast<std::int64_t>(outcome));
    if (outcome == Outcome::Faulted) {
        journal_.record(Mark::FaultReported, job);
    }
    // The frames go back now, not at the next step and not at shutdown. When
    // the settle happened inside a resume the release waits only until control
    // is back out of that resume, so no frame is destroyed underneath itself.
    release_queue_.push_back(job);
    if (resume_depth_ == 0) {
        flush_releases();
    }
}
