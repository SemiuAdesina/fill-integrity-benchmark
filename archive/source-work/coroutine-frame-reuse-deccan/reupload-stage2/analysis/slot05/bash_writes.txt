#!/bin/bash
# $1 = binary, $2 = expected marker, $3 = wanted clean runs
bin=$1; marker=$2; want=${3:-3}
clean=0; tries=0
while [ $clean -lt $want ] && [ $tries -lt 40 ]; do
  tries=$((tries+1))
  ASAN_OPTIONS=detect_leaks=1 "$bin" > /tmp/rs.txt 2>&1
  rc=$?
  if [ $rc -eq 0 ] && grep -q "$marker" /tmp/rs.txt && ! grep -q "ERROR: AddressSanitizer\|runtime error\|LeakSanitizer" /tmp/rs.txt; then
    clean=$((clean+1))
  elif [ $rc -ne 0 ] && ! grep -q "DEADLYSIGNAL" /tmp/rs.txt; then
    echo "REAL FAILURE rc=$rc"; cat /tmp/rs.txt; exit 1
  fi
done
echo "$bin: $clean clean runs of $want (tries=$tries)"
if [ $clean -lt $want ]; then echo "could not get clean runs"; tail -5 /tmp/rs.txt; exit 1; fi
EOF

#include "roundel/runtime.hpp"
#include <cstdio>
#include <string>
#include <vector>
using namespace roundel;

Duty leaf() { co_return co_await park(); }
Duty t_inner() { co_return co_await park(); }
Duty t_middle() { co_return co_await t_inner() + 1; }
Duty t_outer() { co_return co_await t_middle() + 1; }
Duty park_twice() { std::int64_t a = co_await park(); std::int64_t b = co_await park(); co_return a*10+b; }
Duty thrower() { co_await park(); throw std::runtime_error("x"); co_return 0; }
Duty two_kids() { std::int64_t a = co_await leaf(); std::int64_t b = co_await leaf(); co_return a+b; }

struct Tr {
    Conductor& c; std::size_t pos=0; std::string s;
    explicit Tr(Conductor& cc):c(cc){}
    void flush(std::uint64_t now){
        const auto& es=c.journal().entries();
        for(;pos<es.size();++pos){const Entry&e=es[pos];
            switch(e.mark){
            case Mark::TicketAccepted: s+="A"+std::to_string(e.job)+"@"+std::to_string(now)+";";break;
            case Mark::TicketRejected: s+="R"+std::to_string(e.job)+"@"+std::to_string(now)+";";break;
            case Mark::ChildDestroyed: s+="C"+std::to_string(e.job)+";";break;
            case Mark::ParentContinued: s+="P"+std::to_string(e.job)+":"+std::to_string(e.detail)+";";break;
            case Mark::JobSettled: s+="S"+std::to_string(e.job)+"="+
                (e.detail==0?"D":(e.detail==1?"X":"F"))+";";break;
            default:break;}}
    }
    std::string all(){flush(c.now()); if(!s.empty()&&s.back()==';')s.pop_back(); return s;}
};

void show(const char* name, auto&& body) {
    Conductor c{4}; Tr tr{c};
    body(c);
    while (c.step()) tr.flush(c.now());
    std::printf("%-34s %s\n", name, tr.all().c_str());
}

int main(){
    show("3-level ok", [](Conductor& c){ JobId j=c.launch(t_outer()); c.post(j,5); });
    show("3-level cancelled", [](Conductor& c){ JobId j=c.launch(t_outer()); c.cancel(j); });
    show("two children, two tickets", [](Conductor& c){ JobId j=c.launch(two_kids()); c.post(j,1); c.post(j,2); });
    show("child throws", [](Conductor& c){ JobId j=c.launch(t_outer()); c.post_fault(j); });
    show("late fault for settled job", [](Conductor& c){ JobId j=c.launch(leaf()); c.cancel(j); c.post(j,1); });
    show("delays: job2 earlier", [](Conductor& c){ JobId a=c.launch(leaf()); JobId b=c.launch(leaf()); c.post_after(a,1,5); c.post(b,2); });
    show("same ready, lower id first", [](Conductor& c){ JobId a=c.launch(leaf()); JobId b=c.launch(leaf()); c.post(b,2); c.post(a,1); });
    show("repark same frame", [](Conductor& c){ JobId j=c.launch(park_twice()); c.post_after(j,1,5); c.post(j,2); });
    show("cancel parked in child", [](Conductor& c){ JobId j=c.launch(t_outer()); c.post(j,1); c.cancel(j); });
    show("thrower", [](Conductor& c){ JobId j=c.launch(thrower()); c.post(j,1); });
    return 0;
}
EOF
