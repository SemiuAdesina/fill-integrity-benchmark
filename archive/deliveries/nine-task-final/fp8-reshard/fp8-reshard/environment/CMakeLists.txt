cmake_minimum_required(VERSION 3.16)
project(fp8_reshard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(fp8_state src/model.cpp src/restore.cpp)
target_include_directories(fp8_state PUBLIC include)
target_compile_options(fp8_state PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fp8_demo examples/demo.cpp)
target_link_libraries(fp8_demo PRIVATE fp8_state)
