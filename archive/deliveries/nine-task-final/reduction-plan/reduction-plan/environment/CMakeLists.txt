cmake_minimum_required(VERSION 3.20)
project(reduction_plan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(reduction_plan STATIC src/recover.cpp)
target_include_directories(reduction_plan PUBLIC include)
target_compile_options(reduction_plan PRIVATE -Wall -Wextra -Wpedantic)
add_executable(reduction_diagnostic examples/diagnostic.cpp)
target_link_libraries(reduction_diagnostic PRIVATE reduction_plan)
target_compile_options(reduction_diagnostic PRIVATE -Wall -Wextra -Wpedantic)
